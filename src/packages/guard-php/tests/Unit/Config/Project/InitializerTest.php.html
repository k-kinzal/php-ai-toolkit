384    public function testWriteCreatesRecommendedPolicyWithoutOverwritingExistingFiles(): void
385    {
386        $root = sys_get_temp_dir() . '/guard-init-' . uniqid();
387        mkdir($root);
388        file_put_contents($root . '/composer.json', '{"config":{"sort-packages":false}}');
389        (new \Guard\Config\Project\Initializer())->write($root . '/guard.yaml');
390        $config = (new \Guard\Config\ConfigurationLoader())->load($root . '/guard.yaml');
391        $plan = (new \Guard\Execution\Pipeline())->run($config, $root . '/guard.yaml', false);
392        self::assertSame('recommended', $plan->findings[0]->level);
393        self::assertSame('{"config":{"sort-packages":false}}', file_get_contents($root . '/composer.json'));
394        $this->expectException(\Guard\Diagnostic\PolicyException::class);
395        (new \Guard\Config\Project\Initializer())->write($root . '/guard.yaml');
396    }
