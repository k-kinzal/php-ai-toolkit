36        if ($violation->actual !== null) {
37            $output .= sprintf("   actual: %d\n", $violation->actual);

39        if ($violation->limit !== null) {

41        }
42
43        return $output . sprintf("   message: %s\n   action: %s\n", $violation->message, $this->action->action($violation));
