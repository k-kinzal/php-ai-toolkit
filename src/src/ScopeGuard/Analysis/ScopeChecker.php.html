15/**
16 * Checks a scanned project against the scopes its declarations carry.
17 */
18final class ScopeChecker
19{
20    /** @readonly */
21    private ScopeProblemReader $problemReader;
22
23    /** @readonly */
24    private VisibilityScopeResolver $scopeResolver;
25
26    /** @readonly */
27    private ScopeViolationBuilder $violationBuilder;
28
29    /**
30     * Creates the checker from tag diagnosis, scope resolution, and violation building.
31     */
32    public function __construct(
33        ?ScopeProblemReader $problemReader = null,
34        ?VisibilityScopeResolver $scopeResolver = null,
35        ?ScopeViolationBuilder $violationBuilder = null,
36    ) {
37        $this->problemReader = $problemReader ?? new ScopeProblemReader();
38        $this->scopeResolver = $scopeResolver ?? new VisibilityScopeResolver();
39        $this->violationBuilder = $violationBuilder ?? new ScopeViolationBuilder();
40    }
41
42    /**
43     * Returns every violation of the scanned project.
44     *
45     * @return list<Violation>
46     */
47    public function violations(ProjectScan $scan, ExemptNamespaces $exemptNamespaces): array
48    {
49        return array_merge($this->scopeDeclarationViolations($scan), $this->referenceViolations($scan, $exemptNamespaces));
50    }
51
52    /**
53     * Returns the violations of every @visibility tag that cannot be honoured.
54     *
55     * A tag is wrong wherever it is written, so exempt namespaces are checked too.
56     *
57     * @return list<Violation>
58     */
59    public function scopeDeclarationViolations(ProjectScan $scan): array
60    {
61        $violations = [];
62        foreach ($scan->index->declarations() as $declaration) {
63            $violations = array_merge($violations, $this->declarationViolations($declaration));
64        }
65
66        return $violations;
67    }
68
69    /**
70     * Returns the tag violations of one declaration.
71     *
72     * @return list<Violation>
73     */
74    public function declarationViolations(Declaration $declaration): array
75    {
76        $values = $declaration->scope->declaredValues;
77        if ($values === []) {
78            return [];
79        }
80
81        $violations = [];
82        $hasPublic = false;
83        $hasNarrowing = false;
84
85        foreach ($values as $value) {
86            if ($this->scopeResolver->keywordOf($value) === 'public') {
87                $hasPublic = true;
88                continue;
89            }
90
91            $hasNarrowing = true;
92            $problem = $this->problemReader->problem($value, $declaration->namespace);
93            if ($problem !== null) {
94                $violations[] = $this->violationBuilder->invalidScope($declaration, $value, $problem);
95            }
96        }
97
98        if ($hasPublic && $hasNarrowing) {
99            $violations[] = $this->violationBuilder->contradictoryScopes($declaration);
100        }
101
102        return $violations;
103    }
104
105    /**
106     * Returns the violations of every reference that leaves the scope it names.
107     *
108     * @return list<Violation>
109     */
110    public function referenceViolations(ProjectScan $scan, ExemptNamespaces $exemptNamespaces): array
111    {
112        $violations = [];
113        foreach ($scan->references as $reference) {
114            if ($exemptNamespaces->contains($reference->namespace)) {
115                continue;
116            }
117
118            $violation = $this->referenceViolation($scan, $reference);
119            if ($violation !== null) {
120                $violations[] = $violation;
121            }
122        }
123
124        return $violations;
125    }
126
127    /**
128     * Returns the violation of one reference, or null when the reference is in scope.
129     *
130     * A member is never wider than the class-like that declares it, so a reference the
131     * member allows is still checked against the class it belongs to.
132     */
133    public function referenceViolation(ProjectScan $scan, Reference $reference): ?Violation
134    {
135        $member = $reference->memberName === null ? null : $scan->index->memberDeclaration($reference->className, $reference->memberName);
136        if ($member !== null && !$member->scope->permits($reference->namespace)) {
137            return $this->violationBuilder->outOfScope($member, $reference);
138        }
139
140        $class = $scan->index->classDeclaration($reference->className);
141
142        return $class === null || $class->scope->permits($reference->namespace)
143            ? null
144            : $this->violationBuilder->outOfScope($class, $reference);
145    }
146}
