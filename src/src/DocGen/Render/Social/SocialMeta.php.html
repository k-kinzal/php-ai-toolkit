25/**
26 * Renders what a link to one page is previewed from elsewhere.
27 *
28 * A site of relative links needs no address of its own, but a card shown
29 * for a link shared in a chat or on a timeline does: the address is
30 * absolute, and so is the image beside it. Nothing is rendered until a
31 * project says where its site is published, because a card that names
32 * pages and images nobody can fetch is worse than no card at all.
33 */

52    /**
53     * Renders the preview tags of one page, or nothing.
54     *
55     * @param string $pagePath the site path of the page, such as index.html
56     * @param string $title the page title, as the document itself carries it
57     * @param string $description what this page documents, in one sentence
58     */
59    public function render(RenderKit $services, string $pagePath, string $title, string $description): string
60    {
61        $baseUrl = $services->model->baseUrl;
62        if ($baseUrl === null) {
63            return '';
64        }
65
66        $escaper = $services->escaper;
67        $url = $this->url($baseUrl, $pagePath);
68        $heading = sprintf('%s — %s', $title, $services->model->title);
69
70        return sprintf('<link rel="canonical" href="%s">', $escaper->e($url)) . "\n"
71            . $this->tag('name', 'description', $escaper->e($this->summary($description)))
72            . $this->tag('property', 'og:type', 'website')
73            . $this->tag('property', 'og:site_name', $escaper->e($services->model->title))
74            . $this->tag('property', 'og:title', $escaper->e($heading))
75            . $this->tag('property', 'og:description', $escaper->e($this->summary($description)))
76            . $this->tag('property', 'og:url', $escaper->e($url))
77            . $this->image($services, $baseUrl)
78            . $this->tag('name', 'twitter:card', $this->card->supported() ? 'summary_large_image' : 'summary');
79    }

82     * Renders the tags of the card image, or nothing without one.

84    public function image(RenderKit $services, string $baseUrl): string
85    {
86        if (!$this->card->supported()) {
87            return '';
88        }

90        $escaper = $services->escaper;
91
92        return $this->tag('property', 'og:image', $escaper->e($baseUrl . '/' . SocialCard::PATH))
93            . $this->tag('property', 'og:image:width', (string) SocialCard::WIDTH)
94            . $this->tag('property', 'og:image:height', (string) SocialCard::HEIGHT)
95            . $this->tag('property', 'og:image:alt', $escaper->e($services->model->title));
96    }

99     * Renders one meta tag, or nothing when it carries no content.

104    public function tag(string $attribute, string $name, string $content): string
105    {
106        if ($content === '') {
107            return '';
108        }
109
110        return sprintf('<meta %s="%s" content="%s">', $attribute, $name, $content) . "\n";
111    }

113    /**
114     * Returns the absolute address of one page of the site.
115     *
116     * An index page is named by the directory it indexes, because that is
117     * the address readers share and the one a site is linked by.
118     */
119    public function url(string $baseUrl, string $pagePath): string
120    {
121        if ($pagePath === 'index.html') {
122            return $baseUrl . '/';
123        }
124
125        if (str_ends_with($pagePath, '/index.html')) {
126            return $baseUrl . '/' . substr($pagePath, 0, -strlen('index.html'));
127        }
128
129        return $baseUrl . '/' . $pagePath;
130    }

141        }
142
143        return trim(implode('', array_slice($characters, 0, self::SUMMARY_LENGTH))) . '…';
144    }
