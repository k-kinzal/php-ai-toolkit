76    public function testBuildWrapsEveryUsageInOneListItem(): void
77    {
78        $model = new ProjectModel('Demo Docs', '/tmp/none', [], new PackageGraph([]), [], [], new SymbolTable(), new HierarchyIndex(), new UsageIndex(), new TestCaseIndex(), null, [], null, []);
79        $services = new RenderKit($model, new SiteUrl(), new HtmlText(), new PhpHighlighter(), new MarkdownRenderer(), new TypeHtml(), new DoctestExtractor(), new AssertionScanner());
80        $first = new Usage('Demo\Widget', null, 'new', null, null, 'src/Demo/A.php', 3, false);
81        $second = new Usage('Demo\Widget', null, 'type', null, null, 'src/Demo/B.php', 9, false);
82
83        $html = (new UsageListHtml())->build($services, 'index.html', [$first, $second]);
84
85        self::assertStringStartsWith('<ul class="usage-list"><li>', $html);
86        self::assertStringEndsWith('</li></ul>', $html);
87        self::assertSame(2, substr_count($html, '<li>'));
88        self::assertStringContainsString('src/Demo/A.php:3', $html);
89        self::assertStringContainsString('src/Demo/B.php:9', $html);
90        self::assertSame('<ul class="usage-list"></ul>', (new UsageListHtml())->build($services, 'index.html', []));
91    }
