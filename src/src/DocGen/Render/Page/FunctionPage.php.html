88    public function render(RenderKit $services, FunctionDoc $function): string
89    {
90        $pagePath = $services->url->functionPage($function);
91        $templates = [];
92        foreach ($function->docBlock !== null ? $function->docBlock->templates : [] as $template) {
93            $templates[] = $template->name;
94        }
95
96        $context = new TypeRenderContext($pagePath, $function->namespace, $function->useMap, $templates, [], $services->model->symbolTable);
97        $crumbs = [['label' => $function->packageName, 'path' => $services->url->packagePage($function->packageName)]];
98        if ($function->namespace !== '') {
99            $crumbs[] = ['label' => $function->namespace, 'path' => $services->url->namespacePage($function->packageName, $function->namespace)];
100        }
101
102        $crumbs[] = ['label' => $function->shortName . '()', 'path' => null];
103
104        return $this->chrome->page(
105            $services,
106            $pagePath,
107            $function->shortName,
108            $this->descriptions->ofFunction($function),
109            $this->breadcrumb->build($services, $pagePath, $crumbs),
110            $this->sidebar->build($services, $pagePath, new SidebarScope($function->packageName, $function->namespace, $function->fqn, [])),
111            $this->content($services, $pagePath, $function, $context),
112        );
113    }

126            $escaper->e($function->file),
127            $function->startLine,
128        );
129        $html .= '</div>' . "\n";
130        $key = $services->diff->functionKey($function->fqn);
131        $html .= $this->banner->render($services, $services->diff->statusOf($key));
132        $html .= '<pre class="signature"' . $services->diff->attribute($key) . '><code>'
133            . $this->signature->functionSignature($services, $function, $context, $key) . '</code></pre>' . "\n";
134        $html .= $this->docText->render($services, $function->docBlock, $context, $function->shortName . '()');
135        $html .= $this->member->signatureTable($services, $function->parameters, $function->returnType, $function->docBlock, $context, $key);
136        $html .= $this->member->tagExamples($services, $function->docBlock, $function->shortName . '()');
137        $html .= $this->testCaseHtml->section($services, $pagePath, $services->model->testCases->forType($function->fqn));
138
139        return $html . $this->usageList->section($services, $pagePath, 'Called from', $services->model->usages->forType($function->fqn, false), false);
140    }
