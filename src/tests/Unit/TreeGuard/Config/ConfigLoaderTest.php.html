33    public function testLoadParsesTreeYaml(): void
34    {
35        $dir = sys_get_temp_dir() . '/treeguard-config-' . uniqid('', true);
36        mkdir($dir);
37        file_put_contents($dir . '/tree.yaml', <<<'YAML'
38paths:
39  - src
40  - skills
41exclude:
42  - 'src/Generated/*'
43rules:
44  - path: 'src/**'
45    max_files: 25
46    allow: ['*.php']
47    file_case: pascal
48  - path: 'skills/*'
49    require: ['SKILL.md']
50    forbid_empty: true
51report:
52  reporter: json
53  order_by:
54    - rule
55    - path
56YAML);
57
58        $config = (new ConfigLoader())->load($dir . '/tree.yaml');
59
60        self::assertSame($dir, $config->root);
61        self::assertSame(['src', 'skills'], $config->paths);
62        self::assertSame(['src/Generated/*'], $config->exclude);
63        self::assertCount(2, $config->rules);
64        self::assertSame('src/**', $config->rules[0]->path);
65        self::assertSame(25, $config->rules[0]->maxFiles);
66        self::assertSame(['*.php'], $config->rules[0]->allow);
67        self::assertSame('pascal', $config->rules[0]->fileCase);
68        self::assertSame('skills/*', $config->rules[1]->path);
69        self::assertSame(['SKILL.md'], $config->rules[1]->require);
70        self::assertTrue($config->rules[1]->forbidEmpty);
71        self::assertSame('json', $config->report->reporter);
72        self::assertSame(['rule', 'path'], $config->report->orderBy);
73    }
