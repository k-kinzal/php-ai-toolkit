7use PHPStan\Analyser\Scope;
8use PHPStan\Rules\IdentifierRuleError;
9use PHPStan\Rules\Rule;

11/**
12 * Requires a @throws tag for every exception thrown directly in a method body
13 * and not caught within the same method.
14 *
15 * Unlike PHPStan's checked-exception analysis this also applies to unchecked
16 * exceptions: declaring the throw at its origin keeps @throws documentation
17 * accurate and keeps catch analysis sound when "exceptions.implicitThrows"
18 * is disabled. Exceptions raised by called methods do not need to be
19 * propagated manually; only direct throw statements are checked.
20 *
21 * @implements Rule<\PhpParser\Node\Stmt\ClassMethod>
22 */
23final class RequireThrowsTagOnDirectThrowRule implements Rule
24{
25    /** @readonly */
26    private ThrowSiteCollector $throwSiteCollector;
27
28    /** @readonly */
29    private ThrowsDeclarationInspector $throwsDeclarationInspector;
30
31    /** @readonly */
32    private MissingThrowsTagErrorBuilder $missingThrowsTagErrorBuilder;
33
34    /**
35     * Creates the rule from throw-site collection, declaration inspection, and
36     * error building.
37     */
38    public function __construct(
39        ?ThrowSiteCollector $throwSiteCollector = null,
40        ?ThrowsDeclarationInspector $throwsDeclarationInspector = null,
41        ?MissingThrowsTagErrorBuilder $missingThrowsTagErrorBuilder = null,
42    ) {
43        $this->throwSiteCollector = $throwSiteCollector ?? new ThrowSiteCollector();
44        $this->throwsDeclarationInspector = $throwsDeclarationInspector ?? new ThrowsDeclarationInspector();
45        $this->missingThrowsTagErrorBuilder = $missingThrowsTagErrorBuilder ?? new MissingThrowsTagErrorBuilder();
46    }
47
48    /**
49     * @return class-string<\PhpParser\Node\Stmt\ClassMethod>
50     */
51    public function getNodeType(): string
52    {
53        return \PhpParser\Node\Stmt\ClassMethod::class;
54    }
55
56    /**
57     * @param \PhpParser\Node\Stmt\ClassMethod $node
58     * @return list<IdentifierRuleError>
59     */
60    public function processNode(\PhpParser\Node $node, Scope $scope): array
61    {
62        if ($node->stmts === null || $node->stmts === []) {
63            return [];
64        }
65
66        $sites = $this->throwSiteCollector->collect($node->stmts);
67        if ($sites === []) {
68            return [];
69        }
70
71        $classReflection = $scope->getClassReflection();
72        if ($classReflection === null) {
73            return [];
74        }
75
76        $methodName = $node->name->toString();
77        if (!$classReflection->hasNativeMethod($methodName)) {
78            return [];
79        }
80        $declaredThrowType = $classReflection->getNativeMethod($methodName)->getThrowType();
81
82        $errors = [];
83        foreach ($sites as $site) {
84            foreach ($this->throwsDeclarationInspector->uncoveredClassNames($site, $scope, $declaredThrowType) as $thrownClassName) {
85                $errors[] = $this->missingThrowsTagErrorBuilder->undeclaredThrow(
86                    $thrownClassName,
87                    $methodName,
88                    $site->line
89                );
90            }
91        }
92
93        return $errors;
94    }
95}
