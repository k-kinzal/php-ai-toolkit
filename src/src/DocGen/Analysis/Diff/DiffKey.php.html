10/**
11 * Builds the lookup keys under which diff states are recorded.
12 *
13 * Symbol names are compared case-insensitively, exactly as PHP resolves
14 * them, so a class that only changed the case of its name is not reported
15 * as one class added next to one removed.
16 */
17final class DiffKey
18{
19    /**
20     * Member kind of a class constant.
21     */
22    public const CONSTANT = 'constant';
23
24    /**
25     * Member kind of a property.
26     */
27    public const PROPERTY = 'property';
28
29    /**
30     * Member kind of a method.
31     */
32    public const METHOD = 'method';
33
34    /**
35     * Member kind of an enum case.
36     */
37    public const ENUM_CASE = 'case';
38
39    /**
40     * Returns the key of one class, interface, trait, or enum.
41     */
42    public function classLike(string $fqcn): string
43    {
44        return 'c:' . strtolower(ltrim($fqcn, '\\'));
45    }
46
47    /**
48     * Returns the key of the declaration head of one class-like symbol.
49     */
50    public function header(string $fqcn): string
51    {
52        return 'h:' . strtolower(ltrim($fqcn, '\\'));
53    }
54
55    /**
56     * Returns the key of one top-level function.
57     */
58    public function functionSymbol(string $fqn): string
59    {
60        return 'f:' . strtolower(ltrim($fqn, '\\'));
61    }
62
63    /**
64     * Returns the key of one member of a class-like symbol.
65     *
66     * @param string $kind the member kind, such as method or property
67     */
68    public function member(string $fqcn, string $kind, string $name): string
69    {
70        return 'm:' . strtolower(ltrim($fqcn, '\\')) . '::' . $kind . '.' . $name;
71    }
72
73    /**
74     * Returns the key of one parameter of a method or function.
75     *
76     * @param string $ownerKey the member or function key of the declaration
77     */
78    public function parameter(string $ownerKey, string $name): string
79    {
80        return 'p:' . $ownerKey . '#' . $name;
81    }
82
83    /**
84     * Returns the key of the return type of a method or function.
85     *
86     * @param string $ownerKey the member or function key of the declaration
87     */
88    public function returnType(string $ownerKey): string
89    {
90        return 'r:' . $ownerKey;
91    }
92
93    /**
94     * Returns the key of the throws tags of a method or function.
95     *
96     * @param string $ownerKey the member or function key of the declaration
97     */
98    public function throwsTags(string $ownerKey): string
99    {
100        return 't:' . $ownerKey;
101    }
102
103    /**
104     * Returns the key of one rendered Markdown document.
105     */
106    public function document(string $packageName, string $path): string
107    {
108        return 'd:' . $packageName . '/' . $path;
109    }
110
111    /**
112     * Returns the key of one documented package.
113     */
114    public function package(string $packageName): string
115    {
116        return 'k:' . $packageName;
117    }
118
119    /**
120     * Returns the key of one namespace of a package.
121     */
122    public function namespaceName(string $packageName, string $namespace): string
123    {
124        return 'n:' . $packageName . '/' . strtolower($namespace);
125    }
