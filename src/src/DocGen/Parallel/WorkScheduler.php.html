11/**
12 * Splits the work of one generation phase into balanced worker jobs.
13 *
14 * Work items are not equally expensive: one source file can be twenty
15 * times the size of the next, and a job that collected the heaviest of
16 * them would still be running long after every other worker went idle.
17 * Jobs are therefore cut where the accumulated weight of the items crosses
18 * an equal share of the total, which balances them without reordering
19 * anything: every job is a run of consecutive items, so putting the
20 * results of the jobs back together in job order puts the items back in
21 * exactly the order a single process would have worked through them.
22 *
23 * That ordering is the whole point. What the analysis finds and what the
24 * renderer writes both depend on the order the items were worked in, so a
25 * scheme that mixed items between jobs would have to carry each item's
26 * original position through the workers and sort them again on the way
27 * back. Consecutive jobs make that bookkeeping unnecessary.
28 */
29final class WorkScheduler
30{
31    /**
32     * The fewest items a job should hold before another worker is started.
33     *
34     * A worker costs a fork, a serialized result, and a process to wait
35     * for, which is not worth paying for a handful of items.
36     */
37    public const MINIMUM_ITEMS_PER_JOB = 8;
38
39    /** @readonly */
40    private WorkerCount $workers;
41
42    /**
43     * Creates a scheduler from the worker count of this machine.
44     */
45    public function __construct(?WorkerCount $workers = null)
46    {
47        $this->workers = $workers ?? new WorkerCount();
48    }
49
50    /**
51     * Cuts the items of one phase into the jobs its workers will run.
52     *
53     * @template TItem
54     *
55     * @param list<TItem> $items
56     * @param callable(TItem): int $weight the relative cost of one item
57     * @param ?int $workers the count a run asked for, or null for the default
58     *
59     * @return list<list<TItem>>
60     */
61    public function plan(array $items, callable $weight, ?int $workers): array
62    {
63        return $this->schedule($items, $weight, $this->affordableWorkers($this->workers->resolve($workers), count($items)));
64    }
65
66    /**
67     * Cuts the items into one job per worker, balanced by weight.
68     *
69     * @template TItem
70     *
71     * @param list<TItem> $items
72     * @param callable(TItem): int $weight the relative cost of one item
73     * @param int $jobs how many jobs to cut into, at least one
74     *
75     * @return list<list<TItem>> consecutive runs, together the whole list
76     */
77    public function schedule(array $items, callable $weight, int $jobs): array
78    {
79        $count = min(max($jobs, 1), count($items));
80        if ($count < 2) {
81            return $items === [] ? [] : [$items];
82        }
83
84        $weights = $this->weigh($items, $weight);
85        $total = 0;
86        foreach ($weights as $itemWeight) {
87            $total += $itemWeight;
88        }
89
90        $scheduled = [];
91        $job = [];
92        $carried = 0;
93        $left = count($items);
94        foreach ($items as $position => $item) {
95            $job[] = $item;
96            $carried += $weights[$position];
97            $left--;
98            $open = $count - count($scheduled);
99            if ($open > 1 && $left >= $open - 1 && $carried * $count >= $total * (count($scheduled) + 1)) {
100                $scheduled[] = $job;
101                $job = [];
102            }
103        }
104
105        if ($job !== []) {
106            $scheduled[] = $job;
107        }
108
109        return $scheduled;
110    }
111
112    /**
113     * Weighs every item, never lighter than one.
114     *
115     * An item of no weight at all would let a whole run of items fall into
116     * one job while the scheduler believed it had shared them out, so the
117     * count of the items is the floor under whatever the caller measures.
118     *
119     * @template TItem
120     *
121     * @param list<TItem> $items
122     * @param callable(TItem): int $weight
123     *
124     * @return list<int>
125     */
126    public function weigh(array $items, callable $weight): array
127    {
128        $weights = [];
129        foreach ($items as $item) {
130            $weights[] = max($weight($item), 1);
131        }
132
133        return $weights;
134    }
135
136    /**
137     * Returns how many workers one phase should be split across.
138     *
139     * @param int $requested the worker count asked for, such as by --jobs
140     * @param int $items how many items the phase has to work through
141     */
142    public function affordableWorkers(int $requested, int $items): int
143    {
144        $affordable = (int) ($items / self::MINIMUM_ITEMS_PER_JOB);
145
146        return max(min(max($requested, 1), $affordable), 1);
147    }
148}
