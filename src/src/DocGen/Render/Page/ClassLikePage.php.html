89     * Renders one complete class-like page document.
90     */
91    public function render(RenderKit $services, ClassLikeDoc $classLike): string
92    {
93        $pagePath = $services->url->classLikePage($classLike);
94        $context = $this->context($services, $pagePath, $classLike, []);
95        $crumbs = [['label' => $classLike->packageName, 'path' => $services->url->packagePage($classLike->packageName)]];
96        if ($classLike->namespace !== '') {
97            $crumbs[] = ['label' => $classLike->namespace, 'path' => $services->url->namespacePage($classLike->packageName, $classLike->namespace)];
98        }
99
100        $crumbs[] = ['label' => $classLike->shortName, 'path' => null];
101
102        return $this->chrome->page(
103            $services,
104            $pagePath,
105            $classLike->shortName,
106            $this->descriptions->ofClassLike($classLike),
107            $this->breadcrumb->build($services, $pagePath, $crumbs),
108            $this->sidebar->build($services, $pagePath, new SidebarScope(
109                $classLike->packageName,
110                $classLike->namespace,
111                $classLike->fqcn,
112                $this->sections($services, $classLike),
113            )),
114            $this->content($services, $pagePath, $classLike, $context),
115        );
116    }

118    /**
119     * Lists the sections this page will render, in page order.
120     *
121     * Every anchor carries the state of the section it points at, so the
122     * navigation of a page narrows exactly as the page itself does.
123     *
124     * @return list<array{id: string, label: string, status: string}>
125     */
126    public function sections(RenderKit $services, ClassLikeDoc $classLike): array
127    {
128        $constants = $this->visibleMembers($classLike->constants);
129        $properties = $this->visibleMembers($classLike->properties);
130        $methods = $this->visibleMembers($classLike->methods);
131        $candidates = [
132            ['id' => 'aliases', 'label' => 'Type aliases', 'present' => ($classLike->docBlock !== null ? $classLike->docBlock->aliases : []) !== [], 'status' => $services->diff->headerStatus($classLike->fqcn)],
133            ['id' => 'cases', 'label' => 'Cases', 'present' => $classLike->enumCases !== [], 'status' => $this->sectionStatus($services, $classLike, DiffKey::ENUM_CASE, $classLike->enumCases)],
134            ['id' => 'constants', 'label' => 'Constants', 'present' => $constants !== [], 'status' => $this->sectionStatus($services, $classLike, DiffKey::CONSTANT, $constants)],
135            ['id' => 'properties', 'label' => 'Properties', 'present' => $properties !== [], 'status' => $this->sectionStatus($services, $classLike, DiffKey::PROPERTY, $properties)],
136            ['id' => 'methods', 'label' => 'Methods', 'present' => $methods !== [], 'status' => $this->sectionStatus($services, $classLike, DiffKey::METHOD, $methods)],
137            ['id' => 'private-surface', 'label' => 'Private surface', 'present' => $this->privateSurface->members($classLike) !== [], 'status' => $services->diff->combine($this->privateSurface->statuses($services, $classLike))],
138            ['id' => 'test-cases', 'label' => 'Test cases', 'present' => $services->model->testCases->forType($classLike->fqcn) !== [], 'status' => DiffStatus::SAME],
139            ['id' => 'relations', 'label' => 'Relations', 'present' => true, 'status' => DiffStatus::SAME],
140        ];
141
142        $sections = [];
143        foreach ($candidates as $candidate) {
144            if ($candidate['present']) {
145                $sections[] = ['id' => $candidate['id'], 'label' => $candidate['label'], 'status' => $candidate['status']];
146            }
147        }
148
149        return $sections;
150    }

178    public function context(RenderKit $services, string $pagePath, ClassLikeDoc $classLike, array $extraTemplates): TypeRenderContext
179    {
180        $templates = [];
181        foreach (array_merge($classLike->docBlock !== null ? $classLike->docBlock->templates : [], $extraTemplates) as $template) {
182            $templates[] = $template->name;
183        }
184
185        $aliases = [];
186        foreach ($classLike->docBlock !== null ? $classLike->docBlock->aliases : [] as $alias) {
187            $aliases[$alias->name] = '#alias.' . $alias->name;
188        }
189
190        return new TypeRenderContext($pagePath, $classLike->namespace, $classLike->useMap, $templates, $aliases, $services->model->symbolTable);
191    }

193    /**
194     * Renders the page content of one class-like symbol.
195     */
196    public function content(RenderKit $services, string $pagePath, ClassLikeDoc $classLike, TypeRenderContext $context): string
197    {
198        $escaper = $services->escaper;
199        $html = '<div class="symbol-head">';
200        $html .= sprintf('<h1><span class="chip chip-kind k-%s">%s</span>%s</h1>', $classLike->kind, $classLike->kind, $escaper->e($classLike->shortName));
201        $html .= '<div class="symbol-meta">';
202        foreach ($services->model->layerAssignments[strtolower($classLike->fqcn)] ?? [] as $layer) {
203            $html .= sprintf(
204                '<a class="chip chip-layer" href="%s" title="deptrac layer">%s</a>',
205                $escaper->e($services->url->href($pagePath, $services->url->layerPage($classLike->packageName, $layer))),
206                $escaper->e($layer),
207            );
208        }
209
210        $html .= sprintf(
211            '<a class="src-link" href="%s">%s:%d</a>',
212            $escaper->e($services->url->href($pagePath, $services->url->sourcePage($classLike->file)) . '#L' . $classLike->startLine),
213            $escaper->e($classLike->file),
214            $classLike->startLine,
215        );
216        $html .= '</div></div>' . "\n";
217        $html .= $this->banner->render($services, $services->diff->classLikeStatus($classLike->fqcn));
218        $html .= $this->signature->classSignature($services, $classLike, $context);
219        $html .= $this->docText->render($services, $classLike->docBlock, $context);
220        $html .= $this->aliasSection($services, $pagePath, $classLike, $context);
221        $html .= $this->member->tagExamples($services, $classLike->docBlock);
222        $html .= $this->memberSections($services, $pagePath, $classLike, $context);
223        $html .= $this->privateSurface->section($services, $classLike, $context);
224        $html .= $this->testCaseSection($services, $pagePath, $classLike);
225
226        return $html . $this->relations->build($services, $pagePath, $classLike);
227    }

229    /**
230     * Renders the test cases that exercise the whole symbol.
231     */
232    public function testCaseSection(RenderKit $services, string $pagePath, ClassLikeDoc $classLike): string
233    {
234        $testCases = $services->model->testCases->forType($classLike->fqcn);
235        if ($testCases === []) {
236            return '';
237        }
238
239        $dedicated = [];
240        $others = [];
241        foreach ($testCases as $testCase) {
242            if ($this->isDedicatedTest($testCase->testClass, $classLike->shortName)) {
243                $dedicated[] = $testCase;
244            } else {
245                $others[] = $testCase;
246            }
247        }
248
249        return sprintf(
250            '<section%s><h2 id="test-cases">Test cases <span class="count">%d</span><a class="anchor" href="#test-cases">§</a></h2>',
251            $services->diff->unchanged(),
252            count($testCases),
253        )
254            . '<p class="section-note">Test cases that cover or call this symbol, from the coverage report and from the analyzed test sources.</p>'
255            . $this->testCaseHtml->subSection($services, $pagePath, 'Dedicated tests', $dedicated, true)
256            . $this->testCaseHtml->subSection($services, $pagePath, 'Other tests reaching this symbol', $others, false)
257            . '</section>' . "\n";
258    }

260    /**
261     * Reports whether a test class is the dedicated test of a symbol.
262     *
263     * The convention of one test class per source class is what makes a
264     * test the authoritative description of a symbol's behavior, so those
265     * cases are listed before the tests that merely pass through it.
266     */
267    public function isDedicatedTest(string $testClass, string $shortName): bool
268    {
269        $position = strrpos($testClass, '\\');
270
271        return ($position === false ? $testClass : substr($testClass, $position + 1)) === $shortName . 'Test';
272    }

278    {

289            $html .= sprintf(

292                $services->escaper->e($alias->name),
293                $definition,
294            );
295        }
296
297        return $html . '</section>' . "\n";

301     * Renders the case, constant, property, and method sections.
302     */
303    public function memberSections(RenderKit $services, string $pagePath, ClassLikeDoc $classLike, TypeRenderContext $context): string
304    {
305        $html = '';
306        if ($classLike->enumCases !== []) {
307            $html .= '<section' . $this->sectionMark($services, $classLike, DiffKey::ENUM_CASE, $classLike->enumCases)
308                . '><h2 id="cases">Cases<a class="anchor" href="#cases">§</a></h2>';
309            foreach ($classLike->enumCases as $case) {
310                $html .= $this->member->enumCase($services, $pagePath, $classLike, $case, $context);
311            }
312
313            $html .= '</section>' . "\n";
314        }
315
316        $visible = static fn (string $visibility): bool => $visibility !== 'private';
317        $constants = [];
318        foreach ($classLike->constants as $constant) {
319            if ($visible($constant->visibility)) {
320                $constants[] = $constant;
321            }
322        }
323
324        if ($constants !== []) {
325            $html .= '<section' . $this->sectionMark($services, $classLike, DiffKey::CONSTANT, $constants)
326                . '><h2 id="constants">Constants<a class="anchor" href="#constants">§</a></h2>';
327            foreach ($constants as $constant) {
328                $html .= $this->member->constant($services, $pagePath, $classLike, $constant, $context);
329            }
330
331            $html .= '</section>' . "\n";
332        }
333
334        $properties = [];
335        foreach ($classLike->properties as $property) {
336            if ($visible($property->visibility)) {
337                $properties[] = $property;
338            }
339        }
340
341        if ($properties !== []) {
342            $html .= '<section' . $this->sectionMark($services, $classLike, DiffKey::PROPERTY, $properties)
343                . '><h2 id="properties">Properties<a class="anchor" href="#properties">§</a></h2>';
344            foreach ($properties as $property) {
345                $html .= $this->member->property($services, $pagePath, $classLike, $property, $context);
346            }
347
348            $html .= '</section>' . "\n";
349        }
350
351        return $html . $this->methodSection($services, $pagePath, $classLike);
352    }

355     * Renders the method section with per-method template scopes.
356     */
357    public function methodSection(RenderKit $services, string $pagePath, ClassLikeDoc $classLike): string
358    {
359        $methods = [];
360        foreach ($classLike->methods as $method) {
361            if ($method->visibility !== 'private') {
362                $methods[] = $method;
363            }
364        }
365
366        if ($methods === []) {
367            return '';
368        }
369
370        $html = '<section' . $this->sectionMark($services, $classLike, DiffKey::METHOD, $methods)
371            . '><h2 id="methods">Methods<a class="anchor" href="#methods">§</a></h2>';
372        foreach ($methods as $method) {
373            $methodContext = $this->context(
374                $services,
375                $pagePath,
376                $classLike,
377                $method->docBlock !== null ? $method->docBlock->templates : [],
378            );
379            $html .= $this->member->method($services, $pagePath, $classLike, $method, $methodContext);
380        }
381
382        return $html . '</section>' . "\n";
383    }
