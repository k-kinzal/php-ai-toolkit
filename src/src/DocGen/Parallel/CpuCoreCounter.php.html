21/**
22 * Counts the logical CPU cores available to the current machine.
23 *
24 * The sources are asked in the order of how much they can be trusted, and
25 * the first one that answers wins. Every source may be missing on a given
26 * machine — a container without procfs, a host with shell execution
27 * disabled, Windows — so a machine that answers nothing counts as one core
28 * and the generation simply stays sequential.
29 */
30final class CpuCoreCounter
31{
32    /**
33     * The environment variables that state a core count directly.
34     *
35     * Windows sets NUMBER_OF_PROCESSORS itself, and container runtimes are
36     * commonly configured to export it so a process sees the cores of its
37     * cgroup rather than the cores of the host.
38     *
39     * @var list<string>
40     */
41    public const ENVIRONMENT_VARIABLES = ['DOCGEN_CPU_CORES', 'NUMBER_OF_PROCESSORS'];
42
43    private ?int $cores = null;
44
45    /**
46     * Returns the number of logical cores, at least one.
47     */
48    public function count(): int
49    {
50        if ($this->cores !== null) {
51            return $this->cores;
52        }
53
54        return $this->cores = $this->detect() ?? 1;
55    }
56
57    /**
58     * Asks every source in turn for the core count of this machine.
59     */
60    public function detect(): ?int
61    {
62        foreach (self::ENVIRONMENT_VARIABLES as $variable) {
63            $fromEnvironment = $this->fromValue(getenv($variable));
64            if ($fromEnvironment !== null) {
65                return $fromEnvironment;
66            }
67        }
68
69        if (PHP_OS_FAMILY === 'Windows') {
70            return null;
71        }
72
73        foreach (['nproc 2>/dev/null', 'sysctl -n hw.logicalcpu 2>/dev/null', 'getconf _NPROCESSORS_ONLN 2>/dev/null'] as $command) {
74            $fromCommand = $this->fromCommand($command);
75            if ($fromCommand !== null) {
76                return $fromCommand;
77            }
78        }
79
80        return $this->fromProcCpuInfo('/proc/cpuinfo');
81    }
82
83    /**
84     * Reads a core count from a raw value, or null when it states none.
85     *
86     * @param mixed $value the raw value, as an environment or command gives it
87     */
88    public function fromValue($value): ?int
89    {
90        if (!is_string($value)) {
91            return null;
92        }
93
94        $trimmed = trim($value);
95
96        return preg_match('/^[1-9]\d*$/', $trimmed) === 1 ? (int) $trimmed : null;
97    }
98
99    /**
100     * Runs a command that prints a core count, or returns null.
101     *
102     * Shell execution is unavailable on hosts that disable it, and the
103     * commands themselves exist only on some systems, so anything other
104     * than a plain positive number is treated as no answer at all.
105     */
106    public function fromCommand(string $command): ?int
107    {
108        if (!function_exists('shell_exec')) {
109            return null;
110        }
111
112        return $this->fromValue(@shell_exec($command));
113    }
114
115    /**
116     * Counts the processor entries of a Linux cpuinfo file.
117     */
118    public function fromProcCpuInfo(string $path): ?int
119    {
120        if (!@is_readable($path)) {
121            return null;
122        }
123
124        $contents = @file_get_contents($path);
125        if ($contents === false) {
126            return null;
127        }
128
129        $matches = [];
130        $found = preg_match_all('/^processor\s*:/mi', $contents, $matches);
131
132        return $found !== false && $found > 0 ? count($matches[0]) : null;
133    }
134}
