42    public function testSendAnswersASystemOneRequestInJevShape(): void
43    {
44        $transport = new FakeJevTransport(new AnswerScript(static fn (Question $question, array $subject): ?Answer => $question->type === Question::PREDICATE ? Answer::predicate($subject['name'] === 'bad' ? 0.1 : 0.9) : null), 'jev-9', new RequestBody());
45        $body = '{"model":"jev-latest","state":{"name":"bad"},"questions":{"p":{"type":"noul","instructions":"P?"},"c":{"type":"choice","instructions":"C?","criteria":{"a":"A","b":"B"}},"s":{"type":"score","instructions":"S?","criteria":["low","high"]}}}';
46        $responses = $transport->send([new HttpRequest('POST', 'https://api.typesafe.ai/v1/systemone', [], $body)]);
