31     */
32    public function __construct(
33        ?DirectoryTreeScanner $scanner = null,
34        ?DirectoryPatternMatcher $patternMatcher = null,
35        ?DirectoryRuleInspector $ruleInspector = null,
36    ) {
37        $this->scanner = $scanner ?? new DirectoryTreeScanner();
38        $this->patternMatcher = $patternMatcher ?? new DirectoryPatternMatcher();
39        $this->ruleInspector = $ruleInspector ?? new DirectoryRuleInspector();
40    }
41
42    /**
43     * Analyzes the configured tree and returns listings and violations.
44     */
45    public function analyze(TreeGuardConfig $config): AnalysisResult
46    {
47        $listings = $this->scanner->scan($config);
48        $violations = [];
49
50        foreach ($config->rules as $rule) {
51            foreach ($listings as $listing) {
52                if ($this->patternMatcher->matches($rule->path, $listing->relativePath)) {
53                    $violations = array_merge($violations, $this->ruleInspector->inspect($rule, $listing, $listings));
54                }
55            }
56        }
57
58        return new AnalysisResult($listings, $violations);
59    }
