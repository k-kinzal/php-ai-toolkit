7/**
8 * The four states one documented element can have in a diff.
9 *
10 * A plain constant holder is used instead of a native enum to keep the
11 * toolkit compatible with PHP 8.0.
12 */
13final class DiffStatus
14{
15    /**
16     * State of an element that exists only in the head revision.
17     */
18    public const ADDED = 'added';
19
20    /**
21     * State of an element that exists only in the base revision.
22     */
23    public const REMOVED = 'removed';
24
25    /**
26     * State of an element that exists in both revisions but differs.
27     */
28    public const MODIFIED = 'modified';
29
30    /**
31     * State of an element that is identical in both revisions.
32     */
33    public const SAME = 'same';
34
35    /**
36     * Combines the states of the parts of one element into its own state.
37     *
38     * Parts that disagree make the container modified rather than added or
39     * removed, so a container is only ever called new or gone when every
40     * part of it agrees, and a container that holds one change is never
41     * hidden as unchanged.
42     *
43     * @param list<string> $statuses
44     */
45    public function combine(array $statuses): string
46    {
47        if ($statuses === []) {
48            return self::SAME;
49        }
50
51        $combined = $statuses[0];
52        foreach ($statuses as $status) {
53            if ($status !== $combined) {
54                return self::MODIFIED;
55            }
56        }
57
58        return $combined;
59    }
60}
