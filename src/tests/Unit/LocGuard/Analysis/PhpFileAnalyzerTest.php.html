108    public function testAnalyzeReportsFileFunctionMethodAndComplexityViolations(): void
109    {
110        $file = sys_get_temp_dir() . '/locguard-source-' . uniqid('', true) . '.php';
111        file_put_contents($file, <<<'PHP'
112<?php
113
114function long_function(): void
115{
116    echo '1';
117    echo '2';
118    echo '3';
119}
120
121final class Example
122{
123    public function complexMethod(int $value): void
124    {
125        if ($value > 0) {
126            echo 'positive';
127        }
128        if ($value > 1 && $value < 10) {
129            echo 'range';
130        }
131    }
132}
133PHP);
134
135        $analysis = (new PhpFileAnalyzer())->analyze(
136            $file,
137            'src/Example.php',
138            new LimitConfig(10, 8, 5, 50, 50, 50, 3, 4, 2),
139        );
140
141        self::assertSame('src/Example.php', $analysis->file->path);
142        self::assertSame(21, $analysis->file->physicalLines);
143        self::assertGreaterThan(8, $analysis->file->nonCommentLines);
144        self::assertSame(
145            ['file_lines', 'file_ncloc', 'class_lines', 'function_lines', 'method_lines', 'cyclomatic_complexity'],
146            array_map(static fn ($violation): string => $violation->rule, $analysis->violations),
147        );
148    }
