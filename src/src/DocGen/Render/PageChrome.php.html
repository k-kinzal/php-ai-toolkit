84    /**
85     * Renders the note shown when a page holds none of the changes.
86     *
87     * The note is hidden until the script finds that showing the changes
88     * alone would leave the page empty, which is an answer rather than a
89     * blank page: this page is not part of the comparison.
90     */
91    public function emptyHint(RenderKit $services): string
92    {
93        if (!$services->diff->isActive()) {
94            return '';
95        }
96
97        $escaper = $services->escaper;
98
99        return sprintf(
100            '<p class="diff-empty" id="diff-empty" data-changes="%s" data-off="%s" hidden></p>',
101            $escaper->e(sprintf(
102                'Nothing on this page changed between %s and %s.',
103                $services->diff->baseLabel(),
104                $services->diff->headLabel(),
105            )),
106            $escaper->e(sprintf(
107                'This page documents what %s had and %s no longer has. Switch to Diff to read it.',
108                $services->diff->baseLabel(),
109                $services->diff->headLabel(),
110            )),
111        ) . "\n";
112    }

114    /**
115     * Renders the inline script that restores the display preferences.
116     *
117     * The theme and the diff mode are applied before the document is laid
118     * out, so a page never flashes the wrong colors or the whole file when
119     * the reader asked for the changes alone.
120     */
121    public function bootstrap(RenderKit $services): string
122    {
123        if (!$services->diff->isActive()) {
124            return '<script>try{var t=localStorage.getItem("docgen-theme");if(t){document.documentElement.dataset.theme=t}}catch(e){}</script>';
125        }
126
127        return '<script>try{var t=localStorage.getItem("docgen-theme");if(t){document.documentElement.dataset.theme=t}'
128            . 'document.documentElement.dataset.diffMode=localStorage.getItem("docgen-diff-mode")||"' . self::DEFAULT_DIFF_MODE . '"}catch(e){}</script>';
129    }
