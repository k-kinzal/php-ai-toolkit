16/**
17 * Renders the relationship sections of a class-like page.
18 *
19 * Inheritance is shown as linked rows, production references are grouped by
20 * what they actually do, and test code never appears here: what tests
21 * guarantee is answered by the test case section instead.
22 */
23final class RelationsHtml
24{
25    /**
26     * Reference kinds already covered by the inheritance rows.
27     *
28     * @var list<string>
29     */
30    public const STRUCTURAL_KINDS = ['extends', 'implements', 'use-trait'];
31
32    /** @readonly */
33    private UsageListHtml $usageList;
34
35    /**
36     * Creates a relations renderer from the usage list renderer.
37     */
38    public function __construct(?UsageListHtml $usageList = null)
39    {
40        $this->usageList = $usageList ?? new UsageListHtml();
41    }
42
43    /**
44     * Renders hierarchy relations and the reference groups of one symbol.
45     */
46    public function build(RenderKit $services, string $pagePath, ClassLikeDoc $classLike): string
47    {
48        $rows = $this->hierarchyRows($services, $pagePath, $classLike);
49        $groups = $this->referenceGroups($services, $pagePath, $classLike);
50        if ($rows === '' && $groups === '') {
51            return '';
52        }
53
54        return '<section class="relations"' . $services->diff->unchanged() . '><h2 id="relations">Relations<a class="anchor" href="#relations">§</a></h2>' . "\n"
55            . $rows . $groups . '</section>' . "\n";
56    }
57
58    /**
59     * Renders the inheritance rows of one symbol.
60     */
61    public function hierarchyRows(RenderKit $services, string $pagePath, ClassLikeDoc $classLike): string
62    {
63        $hierarchy = $services->model->hierarchy;
64        $html = '';
65        if ($classLike->kind === ClassLikeKind::INTERFACE_) {
66            $html .= $this->symbolSection($services, $pagePath, 'Implemented by', $hierarchy->implementorsOf($classLike->fqcn));
67            $html .= $this->symbolSection($services, $pagePath, 'Extended by', $hierarchy->interfaceExtendersOf($classLike->fqcn));
68        }
69
70        if ($classLike->kind === ClassLikeKind::CLASS_) {
71            $html .= $this->symbolSection($services, $pagePath, 'Extended by', $hierarchy->subclassesOf($classLike->fqcn));
72        }
73
74        if ($classLike->kind === ClassLikeKind::TRAIT_) {
75            $html .= $this->symbolSection($services, $pagePath, 'Used by', $hierarchy->traitUsersOf($classLike->fqcn));
76        }
77
78        return $html;
79    }
80
81    /**
82     * Renders one collapsible section of related symbols.
83     *
84     * Inheritance uses the same disclosure shape as the reference groups,
85     * so every relation on the page is read the same way.
86     *
87     * @param list<string> $names
88     */
89    public function symbolSection(RenderKit $services, string $pagePath, string $label, array $names): string
90    {
91        if ($names === []) {
92            return '';
93        }
94
95        $html = '<details class="usage-details" open><summary>'
96            . sprintf('%s <span class="count">%d</span>', $services->escaper->e($label), count($names))
97            . '</summary><ul class="usage-list">';
98        foreach ($names as $name) {
99            $target = $services->model->symbolTable->classLike($name);
100            $html .= '<li>' . ($target === null
101                ? sprintf('<code>%s</code>', $services->escaper->e($name))
102                : sprintf(
103                    '<a class="k-%s" href="%s" title="%s">%s</a> <a class="usage-loc" href="%s">%s:%d</a>',
104                    $services->escaper->e($target->kind),
105                    $services->escaper->e($services->url->href($pagePath, $services->url->classLikePage($target))),
106                    $services->escaper->e($target->fqcn),
107                    $services->escaper->e($target->shortName),
108                    $services->escaper->e($services->url->href($pagePath, $services->url->sourcePage($target->file)) . '#L' . $target->startLine),
109                    $services->escaper->e($target->file),
110                    $target->startLine,
111                )) . '</li>';
112        }
113
114        return $html . '</ul></details>' . "\n";
115    }
116
117    /**
118     * Renders the production references of one symbol, grouped by kind.
119     */
120    public function referenceGroups(RenderKit $services, string $pagePath, ClassLikeDoc $classLike): string
121    {
122        $fqcn = $classLike->fqcn;
123        $html = '';
124        foreach ($services->model->usages->forTypeGrouped($fqcn, false) as $kind => $usages) {
125            if (in_array($kind, self::STRUCTURAL_KINDS, true)) {
126                continue;
127            }
128
129            $own = [];
130            foreach ($usages as $usage) {
131                if ($usage->fromFqcn !== $fqcn) {
132                    $own[] = $usage;
133                }
134            }
135
136            $html .= $this->usageList->section(
137                $services,
138                $pagePath,
139                UsageListHtml::KIND_LABELS[$kind] ?? $kind,
140                $own,
141                false,
142            );
143        }
144
145        return $html;
146    }
