18/**
19 * Renders the symbols of one architecture layer.
20 *
21 * Layers come from the project's deptrac configuration, so the page also
22 * states which layers this one may depend on. The namespaces the layer
23 * spans are listed before its symbols: a layer is usually recognised by
24 * the namespaces it owns rather than by the individual classes in it.
25 */

43    /**
44     * Creates a layer page renderer from its collaborators.
45     */
46    public function __construct(
47        ?PageChrome $chrome = null,
48        ?SidebarHtml $sidebar = null,
49        ?BreadcrumbHtml $breadcrumb = null,
50        ?SymbolIndex $symbols = null,
51        ?SymbolListHtml $symbolList = null,
52    ) {
53        $this->chrome = $chrome ?? new PageChrome();
54        $this->sidebar = $sidebar ?? new SidebarHtml();
55        $this->breadcrumb = $breadcrumb ?? new BreadcrumbHtml();
56        $this->symbols = $symbols ?? new SymbolIndex();
57        $this->symbolList = $symbolList ?? new SymbolListHtml();
58    }

60    /**
61     * Renders one complete layer page document.
62     */
63    public function render(RenderKit $services, string $packageName, string $layer): string
64    {
65        $pagePath = $services->url->layerPage($packageName, $layer);
66        $rows = $this->symbols->inLayer($services, $packageName, $layer);
67        $sections = $this->symbolList->sections($rows);
68        if ($rows !== []) {
69            array_unshift($sections, ['id' => 'namespaces', 'label' => 'Namespaces']);
70        }
71
72        $crumbs = [
73            ['label' => $packageName, 'path' => $services->url->packagePage($packageName)],
74            ['label' => 'Layer ' . $layer, 'path' => null],
75        ];
76
77        return $this->chrome->page(
78            $services,
79            $pagePath,
80            'Layer ' . $layer,
81            sprintf('The %s architecture layer of the %s package.', $layer, $packageName),
82            $this->breadcrumb->build($services, $pagePath, $crumbs),
83            $this->sidebar->build($services, $pagePath, new SidebarScope($packageName, null, null, $sections)),
84            $this->content($services, $pagePath, $layer, $rows),
85        );
86    }
87
88    /**
89     * Renders the layer content.
90     *
91     * @param list<SymbolRow> $rows
92     */
93    public function content(RenderKit $services, string $pagePath, string $layer, array $rows): string
94    {
95        $escaper = $services->escaper;
96        $html = sprintf(
97            '<div class="symbol-head"><h1><span class="chip chip-layer">layer</span>%s <span class="count">%d</span></h1></div>',
98            $escaper->e($layer),
99            count($rows),
100        ) . "\n";
101        $html .= $this->dependencyRow($services, $pagePath, $layer);
102        $html .= $this->symbolList->namespaceOverview($services, $pagePath, $rows);
103
104        return $html . $this->symbolList->groups($services, $pagePath, $rows, true);
105    }
106
107    /**
108     * Renders the allowed dependencies of one layer.
109     */
110    public function dependencyRow(RenderKit $services, string $pagePath, string $layer): string
111    {
112        $layers = $services->model->layers;
113        if ($layers === null) {
114            return '';
115        }
116
117        $allowed = $layers->ruleset[$layer] ?? [];
118        if ($allowed === []) {
119            return '<p class="section-note">This layer may not depend on any other layer.</p>' . "\n";
120        }
121
122        $links = [];
123        foreach ($allowed as $name) {
124            $links[] = $services->escaper->e($name);
125        }
126
127        return sprintf(
128            '<p class="section-note">May depend on %s.</p>',
129            implode(', ', $links),
130        ) . "\n";
131    }
