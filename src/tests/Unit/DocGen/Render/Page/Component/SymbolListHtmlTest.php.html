87        $model = new ProjectModel('Demo Docs', '/tmp/none', [], new PackageGraph([]), [], [], new SymbolTable(), new HierarchyIndex(), new UsageIndex(), new TestCaseIndex(), null, [], null, []);

100        );

102            "</table></div></section>\n<section class=\"items\" id=\"classes\"><h2>Classes <span class=\"count\">2</span><a class=\"anchor\" href=\"#classes\">§</a></h2>",
