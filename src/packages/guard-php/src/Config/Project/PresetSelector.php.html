16     * Uses an explicit name list, or the presets that match the project.
17     *
18     * loc.yaml and tree.yaml suppress the metrics and structure presets because those files are copied in full.
19     *
20     * @param ?list<string> $imports

23     * @throws PolicyException when an explicit name is not a shipped preset
24     */
25    public function select(string $root, ?array $imports): array
26    {
27        $names = $this->unique($imports ?? $this->detect($root));
28        foreach ($names as $name) {
29            (new PresetCatalog())->assertKnown($name);
30        }
31        if (is_file($root . '/loc.yaml')) {
32            $names = $this->without($names, 'metrics');
33        }
34        if (is_file($root . '/tree.yaml')) {
35            $names = $this->without($names, 'structure');
36        }
37
38        return $names;
39    }

42     * Detects generic presets from source roots and docs presence, and tool presets from installed configuration.
43     *
44     * @return list<string>
45     * @throws JsonException when composer.json or composer.lock is malformed
46     */
47    public function detect(string $root): array
48    {
49        $names = [];
50        if ((new ToolDetector())->sources($root) !== []) {
51            $names[] = 'metrics';
52            $names[] = 'structure';
53        }
54        if (!is_dir($root . '/docs')) {
55            $names[] = 'disable-doc';
56        }
57        foreach ($this->phpstanPresets($root) as $name) {
58            $names[] = $name;
59        }
60        $phpunit = $this->phpunitPresets($root);
61        foreach ($phpunit as $name) {
62            $names[] = $name;
63        }
64        foreach ($this->doctestPresets($root, $phpunit) as $name) {
65            $names[] = $name;
66        }
67        foreach ($this->toolPresets($root) as $name) {
68            $names[] = $name;
69        }
70        if (is_file($root . '/composer.json')) {
71            $names[] = 'composer';
72        }
73        foreach ($this->documentPresets($root) as $name) {
74            $names[] = $name;
75        }
76
77        return $names;
78    }

123    /**
124     * Returns one PHPUnit preset for each versioned file, or the single resolved major.
125     *
126     * An unversioned file is PHPUnit 13 when versioned files also exist. A multi-major constraint
127     * with no lock and no versioned files is not guessed.
128     *
129     * @return list<string>
130     * @throws JsonException when composer.json or composer.lock is malformed
131     */
132    public function phpunitPresets(string $root): array
133    {
134        $versioned = [
135            'phpunit9' => ['phpunit9.xml', 'phpunit9.xml.dist'],
136            'phpunit10' => ['phpunit10.xml', 'phpunit10.xml.dist'],
137            'phpunit11' => ['phpunit11.xml', 'phpunit11.xml.dist'],
138            'phpunit12' => ['phpunit12.xml', 'phpunit12.xml.dist'],
139        ];
140        $names = [];
141        foreach ($versioned as $name => $candidates) {
142            if ((new Recommendations())->existing($root, $candidates) !== null) {
143                $names[] = $name;
144            }
145        }
146        $plain = (new Recommendations())->existing($root, ['phpunit.xml', 'phpunit.xml.dist']);
147        if ($names !== []) {
148            if ($plain !== null) {
149                $names[] = 'phpunit13';
150            }
151
152            return $names;
153        }
154        if ($plain === null) {
155            return [];
156        }
157        $major = $this->phpunitMajor((new ToolDetector())->packages($root)['phpunit/phpunit'] ?? '');
158        $preset = $major === null ? null : ($major === '13' ? 'phpunit13' : 'phpunit' . $major);
159        if ($preset === null || !in_array($preset, (new PresetCatalog())->names(), true)) {
160            return [];
161        }
162
163        return [$preset];
164    }

166    /**
167     * Returns the only PHPUnit major in a lock version or a single-major constraint.
168     */
169    public function phpunitMajor(string $version): ?string
170    {
171        if (preg_match('/^(\d+)\.\d+/', $version, $locked) === 1) {
172            return $locked[1];
173        }
174        $parts = preg_split('/\s*\|\|\s*/', $version);
175        $majors = [];
176        foreach (is_array($parts) ? $parts : [] as $part) {
177            if (preg_match('/(\d+)/', $part, $match) === 1) {
178                $majors[$match[1]] = $match[1];
179            }
180        }
181
182        return count($majors) === 1 ? array_values($majors)[0] : null;
183    }

187     *
188     * @param list<string> $phpunit
189     * @return list<string>
190     */
191    public function doctestPresets(string $root, array $phpunit): array
192    {
193        $map = [
194            'phpunit9' => 'doctest9',
195            'phpunit10' => 'doctest10',
196            'phpunit11' => 'doctest11',
197            'phpunit12' => 'doctest12',
198            'phpunit13' => 'doctest13',
199        ];
200        $examples = $this->requiresExamples($root);
201        $names = [];
202        foreach ($phpunit as $preset) {
203            $doctest = $map[$preset] ?? null;
204            $file = $doctest === null ? null : (new PresetOverrides())->actual($root, $preset);
205            $source = is_string($file) ? file_get_contents($root . '/' . $file) : false;
206            if ($doctest !== null && $source !== false && (str_contains($source, 'DoctestSuite') || $examples)) {
207                $names[] = $doctest;
208            }
209        }
210
211        return $names;
212    }
