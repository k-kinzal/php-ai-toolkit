24     * Lists the namespaces of one package in sorted order.
25     *
26     * @return list<string>
27     */
28    public function namespacesOf(ProjectModel $model, string $packageName): array
29    {
30        $namespaces = [];
31        foreach ($model->classLikes as $classLike) {
32            if ($classLike->packageName === $packageName && !$classLike->isDev) {
33                $namespaces[$classLike->namespace] = true;
34            }
35        }
36
37        foreach ($model->functions as $function) {
38            if ($function->packageName === $packageName && !$function->isDev) {
39                $namespaces[$function->namespace] = true;
40            }
41        }
42
43        ksort($namespaces);
44
45        return array_keys($namespaces);
46    }
