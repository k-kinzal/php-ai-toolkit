11/**
12 * Sends requests over cURL, several at a time, so a lint run is bounded by the provider and not by one connection.
13 */
14final class CurlTransport implements Transport
15{
16    /**
17     * Creates a transport that keeps at most a number of requests in flight and gives up on each after a timeout.
18     */
19    public function __construct(private int $concurrency = 8, private int $timeout = 60)
20    {
21    }
22
23    /**
24     * Sends every request, at most the configured number at a time, and returns the responses in request order.
25     *
26     * @param list<HttpRequest> $requests
27     * @return list<HttpResponse>
28     * @throws LintException when a request fails before a response arrives
29     */
30    public function send(array $requests): array
31    {
32        $responses = [];
33        foreach (array_chunk($requests, max(1, $this->concurrency), true) as $chunk) {
34            foreach ($this->sendChunk($chunk) as $index => $response) {
35                $responses[$index] = $response;
36            }
37        }
38        ksort($responses);
39
40        return array_values($responses);
41    }
42
43    /**
44     * Sends one group of requests concurrently and returns the responses keyed like the requests.
45     *
46     * @param array<int, HttpRequest> $requests
47     * @return array<int, HttpResponse>
48     * @throws LintException when a request fails before a response arrives
49     */
50    public function sendChunk(array $requests): array
51    {
52        $multi = curl_multi_init();
53        $handles = [];
54        foreach ($requests as $index => $request) {
55            $handles[$index] = $this->handle($request);
56            curl_multi_add_handle($multi, $handles[$index]);
57        }
58        $results = $this->drive($multi);
59        $responses = [];
60        foreach ($handles as $index => $handle) {
61            $responses[$index] = $this->response($handle, $requests[$index], $results[spl_object_id($handle)] ?? CURLE_OK);
62            curl_multi_remove_handle($multi, $handle);
63        }
64        curl_multi_close($multi);
65
66        return $responses;
67    }
68
69    /**
70     * Builds the cURL handle for one request.
71     */
72    public function handle(HttpRequest $request): CurlHandle
73    {
74        $handle = curl_init();
75        $headers = [];
76        foreach ($request->headers as $name => $value) {
77            $headers[] = $name . ': ' . $value;
78        }
79        curl_setopt($handle, CURLOPT_URL, $request->url);
80        curl_setopt($handle, CURLOPT_CUSTOMREQUEST, $request->method);
81        curl_setopt($handle, CURLOPT_POSTFIELDS, $request->body);
82        curl_setopt($handle, CURLOPT_HTTPHEADER, $headers);
83        curl_setopt($handle, CURLOPT_RETURNTRANSFER, true);
84        curl_setopt($handle, CURLOPT_TIMEOUT, $this->timeout);
85        curl_setopt($handle, CURLOPT_CONNECTTIMEOUT, min(10, $this->timeout));
86        curl_setopt($handle, CURLOPT_FOLLOWLOCATION, false);
87
88        return $handle;
89    }
90
91    /**
92     * Runs the multi handle until every transfer has finished and returns each transfer's cURL result code by handle object id.
93     *
94     * @return array<int, int>
95     */
96    public function drive(CurlMultiHandle $multi): array
97    {
98        $results = [];
99        do {
100            $status = curl_multi_exec($multi, $running);
101            while (($message = curl_multi_info_read($multi)) !== false) {
102                $handle = $message['handle'] ?? null;
103                $result = $message['result'] ?? null;
104                if ($handle instanceof CurlHandle && is_int($result)) {
105                    $results[spl_object_id($handle)] = $result;
106                }
107            }
108            if ($running > 0) {
109                curl_multi_select($multi, 1.0);
110            }
111        } while ($running > 0 && $status === CURLM_OK);
112
113        return $results;
114    }
115
116    /**
117     * Reads the response of a finished handle.
118     *
119     * @param int $result the cURL result code of the transfer, CURLE_OK when it completed
120     * @throws LintException when the transfer failed before a response arrived
121     */
122    public function response(CurlHandle $handle, HttpRequest $request, int $result): HttpResponse
123    {
124        $body = curl_multi_getcontent($handle);
125        $status = curl_getinfo($handle, CURLINFO_RESPONSE_CODE);
126        if ($result !== CURLE_OK || $body === null || $status === 0) {
127            $reason = $result === CURLE_OK ? 'no response' : (curl_strerror($result) ?? 'cURL error ' . $result);
128            throw new LintException(sprintf('Could not reach %s: %s. Check the network connection, provider.base_url and provider.timeout in decisions.yaml.', $request->url, $reason));
129        }
130
131        return new HttpResponse($status, $body);
132    }
