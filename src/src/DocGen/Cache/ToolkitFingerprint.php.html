14/**
15 * Identifies the generator a cache entry was written by.
16 *
17 * A cache may only be read back by the generator that wrote it: a changed
18 * parser or a changed page renderer turns the same sources into different
19 * output. The fingerprint is therefore the installed version of this
20 * toolkit and of the libraries it parses and prints with, so upgrading any
21 * of them leaves nothing of the previous cache to be read.
22 *
23 * A generator that changed without being installed again — a checkout of
24 * the toolkit being worked on, or a patched vendor directory — keeps its
25 * fingerprint, because nothing about the installation says otherwise.
26 * Generating with --no-cache or --clear-cache is what answers that case,
27 * and it is what keeps working on the generator from being paid for by
28 * every project that installed it.
29 */
30final class ToolkitFingerprint
31{
32    /**
33     * The installed packages the generated documentation depends on.
34     *
35     * @var list<string>
36     */
37    public const PACKAGES = ['k-kinzal/php-ai-toolkit', 'nikic/php-parser', 'phpstan/phpdoc-parser'];
38
39    private ?string $value = null;
40
41    /**
42     * Returns the fingerprint of this generator, computing it once.
43     */
44    public function value(): string
45    {
46        $parts = [];
47        foreach (self::PACKAGES as $package) {
48            $parts[] = $package . "\0" . $this->versionOf($package);
49        }
50
51        return $this->value ??= hash('sha256', implode("\n", $parts));
52    }
53
54    /**
55     * Returns the installed version of one package, with its reference.
56     *
57     * The reference is what tells two installations of the same branch
58     * apart, so a project that follows a development branch of the toolkit
59     * still stops reading the cache of the commit before.
60     *
61     * A package composer knows nothing about has no version to report: an
62     * installation without composer runtime metadata is consistent with
63     * itself, and a run that needs more than that has --no-cache.
64     */
65    public function versionOf(string $package): string
66    {
67        if (!class_exists(InstalledVersions::class) || !InstalledVersions::isInstalled($package)) {
68            return '';
69        }
70
71        return (string) InstalledVersions::getVersion($package) . '@' . (string) InstalledVersions::getReference($package);
72    }
73}
