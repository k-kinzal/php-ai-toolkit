22/**
23 * Creates and removes the scratch directories of a diff run.
24 *
25 * Removal never follows a symbolic link: a checkout links the installed
26 * dependencies of the project, and following that link would delete the
27 * vendor directory of the working tree.
28 */
29final class TempDirectory
30{
31    /**
32     * Creates a new empty directory below the system temp directory.
33     *
34     * @throws DocGenException when the directory cannot be created
35     */
36    public function create(string $prefix): string
37    {
38        $path = rtrim(sys_get_temp_dir(), '/') . '/' . $prefix . uniqid('', true);
39        if (!@mkdir($path, 0700, true) && !is_dir($path)) {
40            throw new DocGenException(sprintf('Failed to create the temporary directory: %s', $path));
41        }
42
43        return $path;
44    }
45
46    /**
47     * Removes one file, link, or directory tree.
48     *
49     * A link is unlinked instead of followed, and a missing path is not an
50     * error, so cleanup after a failed run is always safe to attempt.
51     */
52    public function remove(string $path): void
53    {
54        if (is_link($path) || is_file($path)) {
55            @unlink($path);
56
57            return;
58        }
59
60        if (!is_dir($path)) {
61            return;
62        }
63
64        $entries = scandir($path);
65        foreach ($entries === false ? [] : $entries as $entry) {
66            if ($entry !== '.' && $entry !== '..') {
67                $this->remove($path . '/' . $entry);
68            }
69        }
70
71        @rmdir($path);
72    }
73}
