15/**
16 * Reads the JSON object a chat model wrote back into a verdict.
17 *
18 * The model's text may be wrapped in a code fence; the fence is removed. An
19 * answer that is missing or that names an option the question does not have
20 * becomes a refusal, so the linter reports nothing for it.
21 */
22final class ChatAnswerReader
23{
24    /**
25     * Turns the model's text into a verdict for an inquiry.
26     *
27     * @param string $content the assistant message
28     * @param string $model the model that answered
29     * @throws LintException when the text is not a JSON object
30     */
31    public function verdict(string $content, string $model, Inquiry $inquiry, string $provider): Verdict
32    {
33        $payload = new Payload($this->decode($content, $provider), $provider . ' answer');
34        $answers = [];
35        foreach ($inquiry->questions as $question) {
36            $answers[$question->name] = $payload->has($question->name) ? $this->answer($payload->nested($question->name), $question) : Answer::refusal();
37        }
38
39        return new Verdict($model, $answers);
40    }
41
42    /**
43     * Decodes the assistant message, stripping a Markdown code fence when the model added one.
44     *
45     * @return array<array-key, mixed>
46     * @throws LintException when the text is not a JSON object
47     */
48    public function decode(string $content, string $provider): array
49    {
50        $text = trim($content);
51        if (preg_match('/^```[a-zA-Z]*\s*(.*?)\s*```$/s', $text, $matches) === 1) {
52            $text = $matches[1];
53        }
54        try {
55            $decoded = json_decode($text, true, 32, JSON_THROW_ON_ERROR);
56        } catch (JsonException $exception) {
57            throw new LintException(sprintf('%s answered with text that is not JSON: %s. Use a model that supports structured JSON output, or set provider.model to one that does.', $provider, $this->excerpt($text)), 0, $exception);
58        }
59        if (!is_array($decoded)) {
60            throw new LintException(sprintf('%s answered with JSON that is not an object: %s. Use a model that follows the answer schema.', $provider, $this->excerpt($text)));
61        }
62
63        return $decoded;
64    }
65
66    /**
67     * Reads one answer in the shape the question's type asked for.
68     *
69     * @throws LintException when a field is missing or malformed
70     */
71    public function answer(Payload $answer, Question $question): Answer
72    {
73        if ($question->type === Question::CHOICE) {
74            $choice = $answer->string('choice');
75            if (!array_key_exists($choice, $question->choices)) {
76                return Answer::refusal();
77            }
78            $confidence = $this->unit($answer->float('confidence'));
79
80            return Answer::choice($choice, $confidence, [$choice => $confidence]);
81        }
82        if ($question->type === Question::SCORE) {
83            $score = min(max(0.0, $answer->float('score')), (float) max(0, count($question->levels) - 1));
84            $confidence = $this->unit($answer->float('confidence'));
85
86            return Answer::score($score, $confidence, [(string) (int) round($score) => $confidence]);
87        }
88
89        return Answer::predicate($this->unit($answer->float('probability')));
90    }
91
92    /**
93     * Clamps a number into the unit interval.
94     */
95    public function unit(float $value): float
96    {
97        return min(1.0, max(0.0, $value));
98    }
99
100    /**
101     * Returns the start of a text for a message.
102     */
103    public function excerpt(string $text): string
104    {
105        $flat = trim((string) preg_replace('/\s+/', ' ', $text));
106
107        return strlen($flat) > 120 ? substr($flat, 0, 120) . '…' : $flat;
108    }
