11/**
12 * Renders the three-way display mode switch of a diff site.
13 *
14 * The three modes answer three different questions: what the code is now,
15 * what changed, and what changed and nothing else. The switch sits next to
16 * the theme toggle because both change how the same page is displayed
17 * rather than which page is displayed.
18 */
19final class DiffModeControl
20{
21    /**
22     * The three display modes with their labels and explanations.
23     *
24     * @var list<array{value: string, label: string, title: string}>
25     */
26    public const MODES = [
27        ['value' => 'off', 'label' => 'Off', 'title' => 'Documentation of the head revision, without diff marks'],
28        ['value' => 'inline', 'label' => 'Diff', 'title' => 'Every page with additions and removals marked'],
29        ['value' => 'changes', 'label' => 'Changes', 'title' => 'Only what the head revision added, changed, or removed'],
30    ];
31
32    /**
33     * Renders the mode switch, or nothing outside a diff site.
34     */
35    public function render(RenderKit $services): string
36    {
37        if (!$services->diff->isActive()) {
38            return '';
39        }
40
41        $escaper = $services->escaper;
42        $html = sprintf(
43            '<span class="diff-range" title="Comparing %s to %s">%s <span class="diff-arrow">→</span> %s</span>',
44            $escaper->e($services->diff->baseLabel()),
45            $escaper->e($services->diff->headLabel()),
46            $escaper->e($services->diff->baseLabel()),
47            $escaper->e($services->diff->headLabel()),
48        );
49        $html .= '<div class="diff-modes" id="diff-modes" role="group" aria-label="Diff display mode">';
50        foreach (self::MODES as $mode) {
51            $html .= sprintf(
52                '<button type="button" class="diff-mode" data-diff-mode="%s" title="%s">%s</button>',
53                $escaper->e($mode['value']),
54                $escaper->e($mode['title']),
55                $escaper->e($mode['label']),
56            );
57        }
58
59        return $html . '</div>';
60    }
61}
