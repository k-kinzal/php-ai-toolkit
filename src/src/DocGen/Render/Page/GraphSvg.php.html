15/**
16 * Renders a layered directed graph as inline SVG.
17 *
18 * Nodes are layered by their longest dependency path, so dependents sit
19 * above their dependencies and the arrows read top-down.
20 */
21final class GraphSvg
22{
23    /** @readonly */
24    private HtmlText $escaper;
25
26    /**
27     * Creates a graph renderer.
28     */
29    public function __construct(?HtmlText $escaper = null)
30    {
31        $this->escaper = $escaper ?? new HtmlText();
32    }

34    /**
35     * Renders one graph.
36     *
37     * @param list<array{id: string, label: string, href: ?string, kind: string}> $nodes
38     * @param list<array{from: string, to: string, kind: string}> $edges
39     */
40    public function render(array $nodes, array $edges): string
41    {
42        if ($nodes === []) {
43            return '';
44        }
45
46        $layers = $this->layers($nodes, $edges);
47        $maxLayer = 0;
48        foreach ($layers as $layer) {
49            $maxLayer = max($maxLayer, $layer);
50        }
51
52        $positions = $this->positions($nodes, $layers, $maxLayer);
53        $width = 0;
54        foreach ($positions as $box) {
55            $width = max($width, $box['x'] + $box['w'] + 16);
56        }
57
58        $height = ($maxLayer + 1) * 84 + 8;
59        $svg = sprintf('<svg class="graph" viewBox="0 0 %d %d" role="img" style="max-width:%dpx">', $width, $height, $width);
60        $svg .= $this->edgesSvg($edges, $positions);
61        foreach ($nodes as $node) {
62            $box = $positions[$node['id']];
63            $inner = sprintf(
64                '<rect class="node node-%s" x="%d" y="%d" width="%d" height="34" rx="7"/><text x="%d" y="%d">%s</text>',
65                $this->escaper->e($node['kind']),
66                $box['x'],
67                $box['y'],
68                $box['w'],
69                $box['x'] + $box['w'] / 2,
70                $box['y'] + 22,
71                $this->escaper->e($node['label']),
72            );
73            $svg .= $node['href'] !== null ? sprintf('<a href="%s">%s</a>', $this->escaper->e($node['href']), $inner) : $inner;
74        }
75
76        return $svg . '</svg>';
77    }

79    /**
80     * Renders the curved edge paths between positioned nodes.
81     *
82     * @param list<array{from: string, to: string, kind: string}> $edges
83     * @param array<string, array{x: int, y: int, w: int}> $positions
84     */
85    public function edgesSvg(array $edges, array $positions): string
86    {
87        $svg = '';
88        foreach ($edges as $edge) {
89            $from = $positions[$edge['from']] ?? null;
90            $to = $positions[$edge['to']] ?? null;
91            if ($from === null || $to === null) {
92                continue;
93            }
94
95            $x1 = $from['x'] + $from['w'] / 2;
96            $y1 = $from['y'] + 34;
97            $x2 = $to['x'] + $to['w'] / 2;
98            $y2 = $to['y'];
99            $svg .= sprintf(
100                '<path class="edge edge-%s" d="M %.1f %.1f C %.1f %.1f, %.1f %.1f, %.1f %.1f"/>',
101                $this->escaper->e($edge['kind']),
102                $x1,
103                $y1,
104                $x1,
105                $y1 + 30,
106                $x2,
107                $y2 - 30,
108                $x2,
109                $y2 - 3,
110            );
111            $svg .= sprintf('<circle class="edge-tip edge-%s" cx="%.1f" cy="%.1f" r="2.6"/>', $this->escaper->e($edge['kind']), $x2, $y2 - 2);
112        }
113
114        return $svg;
115    }

117    /**
118     * Computes the layer of every node from its longest outgoing path.
119     *
120     * @param list<array{id: string, label: string, href: ?string, kind: string}> $nodes
121     * @param list<array{from: string, to: string, kind: string}> $edges
122     *
123     * @return array<string, int>
124     */
125    public function layers(array $nodes, array $edges): array
126    {
127        $out = [];
128        foreach ($edges as $edge) {
129            $out[$edge['from']][] = $edge['to'];
130        }
131
132        $layers = [];
133        foreach ($nodes as $node) {
134            $layers[$node['id']] = $this->longestPath($node['id'], $out, $layers, []);
135        }
136
137        return $layers;
138    }
139
140    /**
141     * Computes the longest path length from one node, guarding cycles.
142     *
143     * @param array<string, list<string>> $out
144     * @param array<string, int> $known
145     * @param array<string, bool> $visiting
146     */
147    public function longestPath(string $id, array $out, array $known, array $visiting): int
148    {
149        if (isset($known[$id])) {
150            return $known[$id];
151        }
152
153        if (isset($visiting[$id])) {
154            return 0;
155        }
156
157        $visiting[$id] = true;
158        $depth = 0;
159        foreach ($out[$id] ?? [] as $next) {
160            $depth = max($depth, 1 + $this->longestPath($next, $out, $known, $visiting));
161        }
162
163        return $depth;
164    }

166    /**
167     * Computes node boxes, centering each layer horizontally.
168     *
169     * @param list<array{id: string, label: string, href: ?string, kind: string}> $nodes
170     * @param array<string, int> $layers
171     *
172     * @return array<string, array{x: int, y: int, w: int}>
173     */
174    public function positions(array $nodes, array $layers, int $maxLayer): array
175    {
176        $rows = [];
177        foreach ($nodes as $node) {
178            $rows[$layers[$node['id']]][] = $node;
179        }
180
181        $rowWidths = [];
182        $totalWidth = 0;
183        foreach ($rows as $layer => $rowNodes) {
184            usort($rowNodes, static fn (array $a, array $b): int => $a['label'] <=> $b['label']);
185            $rows[$layer] = $rowNodes;
186            $width = 0;
187            foreach ($rowNodes as $node) {
188                $width += max(64, 8 * strlen($node['label']) + 26) + 22;
189            }
190
191            $rowWidths[$layer] = $width - 22;
192            $totalWidth = max($totalWidth, $rowWidths[$layer]);
193        }
194
195        $positions = [];
196        foreach ($rows as $layer => $rowNodes) {
197            $x = (int) (8 + ($totalWidth - $rowWidths[$layer]) / 2);
198            $y = ($maxLayer - $layer) * 84 + 8;
199            foreach ($rowNodes as $node) {
200                $width = max(64, 8 * strlen($node['label']) + 26);
201                $positions[$node['id']] = ['x' => $x, 'y' => $y, 'w' => $width];
202                $x += $width + 22;
203            }
204        }
205
206        return $positions;
207    }
