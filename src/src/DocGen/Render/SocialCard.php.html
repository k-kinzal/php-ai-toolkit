31/**
32 * Draws the image a link to the site is previewed with elsewhere.
33 *
34 * A link shared in a chat or on a timeline is rendered from the tags of
35 * the page it points at, and an image is the part of that card a reader
36 * sees first. The image is drawn from the project rather than shipped
37 * with the generator, so every site previews as itself, and it is drawn
38 * in the palette of the site it belongs to.
39 *
40 * Drawing needs the gd extension with FreeType support. Without it a run
41 * writes no image and the pages carry no image tag, because a card that
42 * names an image the site does not serve is worse than a card without
43 * one.
44 */
45final class SocialCard
46{
47    /**
48     * Where the drawn image is written inside the site.
49     */
50    public const PATH = 'assets/og-image.png';

53     * The image size every social platform scales its cards from.
54     */
55    public const WIDTH = 1200;
56

58     * The image height that gives the 1.91:1 ratio cards are cropped to.
59     */
60    public const HEIGHT = 630;

63     * The margin the card keeps on every side.
64     */
65    public const PADDING = 72;

68     * The band of colors the card is marked with, left to right.

103    /**
104     * Draws one card and returns it as PNG bytes.
105     *
106     * @param string $title the site title, printed as the card headline
107     * @param string $subtitle what the documented project is, printed below it
108     *
109     * @throws DocGenException when the image cannot be drawn
110     */
111    public function render(string $title, string $subtitle): string
112    {
113        $image = $this->image();
114        $this->background($image);
115        $this->heading($image, $title);
116        $this->body($image, $subtitle, $title);
117        $this->footer($image);
118
119        $file = tempnam(sys_get_temp_dir(), 'docgen-card-');
120        if ($file === false) {
121            throw new DocGenException('Could not draw the social preview image: no temporary file could be created for it.');
122        }
123
124        imagepng($image, $file, 9);
125        $png = file_get_contents($file);
126        unlink($file);
127
128        if ($png === false || $png === '') {
129            throw new DocGenException('Could not draw the social preview image: the gd extension encoded no PNG.');
130        }
131
132        return $png;
133    }

135    /**
136     * Creates the canvas one card is drawn on.

140    public function image(): GdImage
141    {
142        $image = imagecreatetruecolor(self::WIDTH, self::HEIGHT);
143        if ($image === false) {
144            throw new DocGenException('Could not draw the social preview image: the gd extension refused an image of 1200x630.');
145        }
146
147        return $image;
148    }

150    /**
151     * Fills the card and prints the hue strip that identifies the site.
152     */
153    public function background(GdImage $image): void
154    {
155        imagefilledrectangle($image, 0, 0, self::WIDTH, self::HEIGHT, $this->color($image, '#0d1117'));
156
157        $width = (int) round(self::WIDTH / count(self::HUES));
158        $x = 0;
159        foreach (self::HUES as $hue) {
160            imagefilledrectangle($image, $x, 0, $x + $width, 12, $this->color($image, $hue));
161            $x += $width;
162        }
163    }

165    /**
166     * Prints the label and the title of the card.
167     */
168    public function heading(GdImage $image, string $title): void
169    {
170        $font = $this->fontPath();
171        imagettftext($image, 20.0, 0.0, self::PADDING, 132, $this->color($image, '#7d8590'), $font, 'DOCUMENTATION');
172
173        $size = $this->text->fit($font, $title, self::WIDTH - self::PADDING * 2, 64, 34);
174        $y = 262;
175        foreach ($this->text->wrap($font, (float) $size, $title, self::WIDTH - self::PADDING * 2, 2) as $line) {
176            imagettftext($image, (float) $size, 0.0, self::PADDING, $y, $this->color($image, '#e6edf3'), $font, $line);
177            $y += (int) round($size * 1.4);
178        }
179    }

181    /**
182     * Prints what the documented project is, below its title.
183     *
184     * @param string $title the title already printed, so the subtitle never repeats it
185     */
186    public function body(GdImage $image, string $subtitle, string $title): void
187    {
188        if ($subtitle === '' || $subtitle === $title) {
189            return;
190        }
191
192        $font = $this->fontPath();
193        $y = 380;
194        foreach ($this->text->wrap($font, 26.0, $subtitle, self::WIDTH - self::PADDING * 2, 2) as $line) {
195            imagettftext($image, 26.0, 0.0, self::PADDING, $y, $this->color($image, '#9198a1'), $font, $line);
196            $y += 44;
197        }
198    }

203    public function footer(GdImage $image): void
204    {
205        imagettftext(
206            $image,
207            22.0,
208            0.0,
209            self::PADDING,
210            self::HEIGHT - self::PADDING,
211            $this->color($image, '#7d8590'),
212            $this->fontPath(),
213            'generated by php-ai-toolkit doc-gen',
214        );
215    }

217    /**
218     * Allocates one color of the design system palette.
219     *
220     * @param string $hex the color as it is written in the stylesheet, such as #79c0ff
221     *
222     * @throws DocGenException when the palette runs out of colors
223     */
224    public function color(GdImage $image, string $hex): int
225    {
226        $color = imagecolorallocate(
227            $image,
228            $this->channel($hex, 1),
229            $this->channel($hex, 3),
230            $this->channel($hex, 5),
231        );
232
233        if ($color === false) {
234            throw new DocGenException('Could not draw the social preview image: the gd extension allocated no color.');
235        }
236
237        return $color;
238    }

240    /**
241     * Reads one channel of a hex color as a byte.
242     *
243     * @param string $hex the color as it is written in the stylesheet
244     * @param int $offset where the channel starts, after the leading hash
245     *
246     * @return int<0, 255>
247     */
248    public function channel(string $hex, int $offset): int
249    {
250        return min(255, max(0, (int) hexdec(substr($hex, $offset, 2))));
251    }
