24/**
25 * The Symfony Console application with the check, fix, rules and init commands.
26 *
27 * Help, version, verbosity, colour and shell completion behave as in every
28 * other Symfony Console tool. guard alone runs check, guard --help lists
29 * the commands, and a command name guard does not know exits 2.
30 */
31final class GuardConsole extends ConsoleApplication

42    {
43        parent::__construct('guard', self::VERSION);
44        $this->setAutoExit(false);
45        $this->setCommandLoader(new FactoryCommandLoader([
46            'baseline' => static fn (): BaselineCommand => new BaselineCommand($directory, $registry),
47            'check' => static fn (): CheckCommand => new CheckCommand($directory, $registry),
48            'fix' => static fn (): FixCommand => new FixCommand($directory, $registry),
49            'rules' => static fn (): RulesCommand => new RulesCommand($directory),
50            'init' => static fn (): InitCommand => new InitCommand($directory),

64    /**
65     * Runs the command the input names.
66     *
67     * The input and output are configured here as well as in run(), so a
68     * caller that runs the console in-process still gets --quiet, --verbose
69     * and --no-ansi honoured. The verbosity Symfony Console leaves in the
70     * environment for child processes is put back afterwards, so one
71     * in-process run does not silence the next.
72     */
73    #[Override]
74    public function doRun(InputInterface $input, OutputInterface $output): int
75    {
76        $shellVerbosity = [getenv('SHELL_VERBOSITY'), $_ENV['SHELL_VERBOSITY'] ?? null, $_SERVER['SHELL_VERBOSITY'] ?? null];
77        $this->configureIO($input, $output);
78        try {
79            $name = $this->getCommandName($input);
80            if ($name === null && $input->hasParameterOption(['--help', '-h'], true)) {
81                return parent::doRun(new ArrayInput(['command' => 'list']), $output);
82            }
83            $problem = $name === null ? null : $this->unknownCommand($name);
84            if ($problem !== null) {
85                $errorOutput = $output instanceof ConsoleOutputInterface ? $output->getErrorOutput() : $output;
86                $errorOutput->writeln(['Guard error: ' . $problem, 'Run "guard list" to see the available commands.'], OutputInterface::OUTPUT_RAW | OutputInterface::VERBOSITY_QUIET);
87
88                return GuardCommand::INVALID;
89            }
90
91            return parent::doRun($input, $output);
92        } finally {
93            $this->restoreShellVerbosity(...$shellVerbosity);
94        }
95    }

97    /**
98     * Explains why a command name names no single command, or returns null when it does.
99     *
100     * A name may be cut short as long as only one command starts with it,
101     * such as ch for check.
102     */
103    public function unknownCommand(string $name): ?string
104    {
105        $names = array_keys($this->all());
106        if (in_array($name, $names, true)) {
107            return null;
108        }
109        $matches = array_values(array_filter($names, static fn (string $command): bool => str_starts_with($command, $name) && !str_starts_with($command, '_')));
110        if (count($matches) === 1) {
111            return null;
112        }
113        if ($matches !== []) {
114            sort($matches);
115
116            return sprintf('Command "%s" is ambiguous. Name one of: %s.', $name, implode(', ', $matches));
117        }
118        $similar = array_values(array_filter($names, static fn (string $command): bool => !str_starts_with($command, '_') && levenshtein($name, $command) <= max(2, intdiv(strlen($name), 3))));
119
120        return sprintf('Command "%s" is not defined.', $name) . ($similar === [] ? '' : sprintf(' Did you mean %s?', implode(' or ', $similar)));
121    }

137    /**
138     * Puts the shell verbosity of the environment back to what it was.
139     *
140     * @param false|string $process the process environment value, or false when unset
141     * @param mixed $env the $_ENV value, or null when unset
142     * @param mixed $server the $_SERVER value, or null when unset
143     */
144    public function restoreShellVerbosity($process, $env, $server): void
145    {
146        putenv($process === false ? 'SHELL_VERBOSITY' : 'SHELL_VERBOSITY=' . $process);
147        unset($_ENV['SHELL_VERBOSITY'], $_SERVER['SHELL_VERBOSITY']);
148        if ($env !== null) {
149            $_ENV['SHELL_VERBOSITY'] = $env;
150        }
151        if ($server !== null) {
152            $_SERVER['SHELL_VERBOSITY'] = $server;
153        }
154    }
