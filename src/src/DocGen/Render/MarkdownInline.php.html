14/**
15 * Renders inline Markdown constructs inside one line of text.
16 *
17 * Code spans are protected before escaping so their contents are shown
18 * verbatim; only code, emphasis, and links are supported by design.
19 *
20 * An image is reduced to its alt text: the generated site is meant to be
21 * self-contained, so a README badge must not turn into a request to a
22 * remote host, and the alt text carries the same information.
23 *
24 * A link to an absolute URL or to a page anchor becomes an anchor element,
25 * and a relative path is offered to the link resolver, which turns a link
26 * to a document of the same repository into a link to its rendered page.
27 * Every remaining target has no counterpart in the generated site, so its
28 * text is rendered plainly with the target kept in the title attribute
29 * instead of being emitted as a broken link or, worse, as raw Markdown.
30 */
31final class MarkdownInline
32{
33    /** @readonly */
34    private ?MarkdownLinks $links;
35
36    /**
37     * Creates an inline renderer, optionally resolving document links.
38     */
39    public function __construct(?MarkdownLinks $links = null)
40    {
41        $this->links = $links;
42    }
43
44    /**
45     * Renders inline code, emphasis, and links of one text fragment.
46     */
47    public function render(string $text): string
48    {
49        $escaper = new HtmlText();
50        $links = $this->links;
51        $codes = [];
52        $protected = preg_replace_callback('/`([^`]+)`/', static function (array $match) use (&$codes, $escaper): string {
53            $codes[] = '<code>' . $escaper->e($match[1]) . '</code>';
54
55            return "\x1A" . (count($codes) - 1) . "\x1A";
56        }, $text) ?? $text;
57
58        $html = $escaper->e($protected);
59        $html = preg_replace('/(?<=^|\s)\*\*(\S(?:[^*]*\S)?)\*\*(?=$|[\s.,:;!?)])/', '<strong>$1</strong>', $html) ?? $html;
60        $html = preg_replace('/(?<=^|\s)\*([^*\s][^*]*)\*(?=$|[\s.,:;!?)])/', '<em>$1</em>', $html) ?? $html;
61        $html = preg_replace_callback('/!\[([^\]]*)\]\(((?:[^()\s]|\([^()\s]*\))+)\)/', static function (array $match): string {
62            return $match[1] === ''
63                ? ''
64                : sprintf('<span class="md-target" title="%s">%s</span>', $match[2], $match[1]);
65        }, $html) ?? $html;
66        $html = preg_replace_callback('/\[([^\]]+)\]\(((?:[^()\s]|\([^()\s]*\))+)\)/', static function (array $match) use ($links, $escaper): string {
67            if (preg_match('#^(https?://|\#)#', $match[2]) === 1) {
68                return sprintf('<a href="%s">%s</a>', $match[2], $match[1]);
69            }
70
71            $href = $links !== null ? $links->resolve($match[2]) : null;
72            if ($href !== null) {
73                return sprintf('<a href="%s">%s</a>', $escaper->e($href), $match[1]);
74            }
75
76            return sprintf('<span class="md-target" title="%s">%s</span>', $match[2], $match[1]);
77        }, $html) ?? $html;
78
79        foreach ($codes as $position => $code) {
80            $html = str_replace("\x1A" . $position . "\x1A", $code, $html);
81        }
82
83        return $html;
84    }
85}
