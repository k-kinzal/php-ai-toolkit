16/**
17 * Remembers which page of a site was written from which inputs.
18 *
19 * A page is left alone when everything it was rendered from is unchanged
20 * and the file it was written to is still the file this cache wrote. The
21 * second half of that condition is what keeps the cache honest: a site
22 * somebody emptied, or a page somebody edited, is rendered again rather
23 * than declared up to date.
24 *
25 * One cache serves one output directory, because a page is only up to date
26 * relative to the site it belongs to.
27 */
28final class RenderCache
29{
30    /**
31     * Name prefix of the page cache files below the cache directory.
32     */
33    public const FILE_PREFIX = 'pages-';
34
35    /** @readonly */
36    private string $path;
37
38    /** @readonly */
39    private CacheStore $store;
40
41    /** @var array<string, array{signature: string, size: int}> */
42    private array $pages = [];
43
44    private int $rendered = 0;
45
46    private int $reused = 0;
47
48    /**
49     * Creates the page cache of one output directory.
50     */
51    public function __construct(string $directory, string $outputRoot, ?CacheStore $store = null)
52    {
53        $this->path = $directory . '/' . self::FILE_PREFIX . substr(hash('sha256', $outputRoot), 0, 16) . '.cache';
54        $this->store = $store ?? new CacheStore();
55    }
56
57    /**
58     * Reads the pages written by the previous run of this output directory.
59     *
60     * This is called before the workers are forked, so every worker
61     * inherits the pages instead of reading the cache file again.
62     */
63    public function load(): void
64    {
65        $data = $this->store->read($this->path);
66        $pages = $data['pages'] ?? null;
67        if (!is_array($pages)) {
68            return;
69        }
70
71        foreach ($pages as $path => $page) {
72            $valid = $this->page($page);
73            if (is_string($path) && $valid !== null) {
74                $this->pages[$path] = $valid;
75            }
76        }
77    }
78
79    /**
80     * Reports whether one page is already on disk as it would be written.
81     *
82     * The file is asked about rather than remembered: the answer has to
83     * describe the output directory as it is now, so the stat cache of
84     * this process is dropped for the file before it is read, and a page
85     * something else rewrote is never reported as already written.
86     */
87    public function isFresh(string $outputRoot, string $path, string $signature): bool
88    {
89        $page = $this->pages[$path] ?? null;
90        if ($page === null || $page['signature'] !== $signature) {
91            return false;
92        }
93
94        $file = $outputRoot . '/' . $path;
95        clearstatcache(true, $file);
96
97        return is_file($file) && @filesize($file) === $page['size'];
98    }
99
100    /**
101     * Returns the byte count remembered for one page.
102     */
103    public function sizeOf(string $path): int
104    {
105        return $this->pages[$path]['size'] ?? 0;
106    }
107
108    /**
109     * Takes the records of a finished run as the new state of the site.
110     *
111     * Pages the run did not report are pages the site no longer has, so
112     * their files are removed: a site is what this run says it is, not what
113     * it says plus whatever an earlier run happened to leave behind.
114     *
115     * @param list<PageRecord> $records
116     *
117     * @return bool whether the cache file was written
118     */
119    public function record(string $outputRoot, array $records): bool
120    {
121        $pages = [];
122        foreach ($records as $record) {
123            $pages[$record->path] = ['signature' => $record->signature, 'size' => $record->size];
124            if ($record->rendered) {
125                $this->rendered++;
126            } else {
127                $this->reused++;
128            }
129        }
130
131        foreach ($this->pages as $path => $page) {
132            if (!isset($pages[$path])) {
133                @unlink($outputRoot . '/' . $path);
134            }
135        }
136
137        $this->pages = $pages;
138
139        return $this->store->write($this->path, ['pages' => $pages]);
140    }
141
142    /**
143     * Reads one stored page as the record of a written page it must be.
144     *
145     * @param mixed $page
146     *
147     * @return ?array{signature: string, size: int}
148     */
149    public function page($page): ?array
150    {
151        if (!is_array($page) || !isset($page['signature'], $page['size'])) {
152            return null;
153        }
154
155        $signature = $page['signature'];
156        $size = $page['size'];
157
158        return is_string($signature) && is_int($size) ? ['signature' => $signature, 'size' => $size] : null;
159    }
160
161    /**
162     * Returns how many pages this run rendered anew.
163     */
164    public function rendered(): int
165    {
166        return $this->rendered;
167    }
168
169    /**
170     * Returns how many pages this run left as they were.
171     */
172    public function reused(): int
173    {
174        return $this->reused;
175    }
176}
