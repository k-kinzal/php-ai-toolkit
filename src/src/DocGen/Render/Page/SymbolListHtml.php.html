17/**
18 * Renders kind-grouped symbol listings.
19 *
20 * Namespace, layer, and all-item pages share this listing so a symbol is
21 * always presented the same way, whichever route led to it.
22 */

40    /**
41     * Renders one section per kind, each anchored by its kind name.
42     *
43     * @param list<SymbolRow> $rows
44     */
45    public function groups(RenderKit $services, string $pagePath, array $rows, bool $withNamespace = false): string
46    {
47        $html = '';
48        foreach ($this->symbols->byKind($rows) as $kind => $kindRows) {
49            $html .= sprintf(
50                '<section class="items"%s id="%s"><h2>%s <span class="count">%d</span><a class="anchor" href="#%s">§</a></h2>',
51                $services->diff->combined($this->statuses($kindRows)),
52                $services->escaper->e(SymbolIndex::KIND_ANCHORS[$kind] ?? $kind),
53                $services->escaper->e(SymbolIndex::KIND_LABELS[$kind]),
54                count($kindRows),
55                $services->escaper->e(SymbolIndex::KIND_ANCHORS[$kind] ?? $kind),
56            );
57            $html .= $this->table($services, $pagePath, $kindRows, $withNamespace) . '</section>' . "\n";
58        }
59
60        return $html;
61    }

63    /**
64     * Renders one table of symbol rows.
65     *
66     * Listings that span namespaces show the namespace of every row, so a
67     * name is never ambiguous; a namespace listing omits the column.
68     *
69     * @param list<SymbolRow> $rows
70     */
71    public function table(RenderKit $services, string $pagePath, array $rows, bool $withNamespace = false): string
72    {
73        $html = '<div class="table-wrap"><table class="item-table">';
74        foreach ($rows as $row) {
75            $html .= sprintf(
76                '<tr%s><td><a class="item-name k-%s" href="%s">%s</a></td>%s<td class="item-summary">%s</td></tr>',
77                $services->diff->mark($row->status),
78                $services->escaper->e($row->kind),
79                $services->escaper->e($services->url->href($pagePath, $row->page)),
80                $services->escaper->e($row->name),
81                $withNamespace ? $this->namespaceCell($services, $pagePath, $row) : '',
82                $this->inline->render($row->summary),
83            );
84        }
85
86        return $html . '</table></div>';
87    }

89    /**
90     * Renders the namespace overview table of a listing.
91     *
92     * A listing that spans namespaces opens with the namespaces it covers,
93     * so the shape of the scope is read before the individual symbols.
94     *
95     * @param list<SymbolRow> $rows
96     */
97    public function namespaceOverview(RenderKit $services, string $pagePath, array $rows): string
98    {
99        $groups = [];
100        foreach ($rows as $row) {
101            $groups[$row->namespace][] = $row;
102        }
103
104        if ($groups === []) {
105            return '';
106        }
107
108        ksort($groups);
109        $html = '<section' . $services->diff->combined($this->statuses($rows))
110            . '><h2 id="namespaces">Namespaces<a class="anchor" href="#namespaces">§</a></h2><div class="table-wrap"><table class="symbol-table">';
111        foreach ($groups as $groupRows) {
112            $html .= $this->namespaceRow($services, $pagePath, $groupRows);
113        }
114
115        return $html . '</table></div></section>' . "\n";
116    }

118    /**
119     * Renders one namespace row with its symbol counts per kind.
120     *
121     * @param non-empty-list<SymbolRow> $rows
122     */
123    public function namespaceRow(RenderKit $services, string $pagePath, array $rows): string
124    {
125        $namespace = $rows[0]->namespace;
126        $packageName = $this->packageOf($services, $rows[0]);
127        $label = $services->escaper->e($namespace === '' ? '(global)' : $namespace);
128
129        return sprintf(
130            '<tr%s><td>%s</td><td class="ns-counts">%s</td></tr>',
131            $services->diff->combined($this->statuses($rows)),
132            $packageName === '' ? $label : sprintf(
133                '<a href="%s">%s</a>',
134                $services->escaper->e($services->url->href($pagePath, $services->url->namespacePage($packageName, $namespace))),
135                $label,
136            ),
137            $this->kindCounts($services, $rows),
138        );
139    }

141    /**
142     * Renders the per-kind symbol counts of one listing.
143     *
144     * @param list<SymbolRow> $rows
145     */
146    public function kindCounts(RenderKit $services, array $rows): string
147    {
148        $html = '';
149        foreach ($this->symbols->byKind($rows) as $kind => $kindRows) {
150            $count = count($kindRows);
151            $html .= sprintf(
152                ' <span class="ns-count k-%s">%d %s</span>',
153                $services->escaper->e($kind),
154                $count,
155                $services->escaper->e($count === 1 ? $kind : strtolower(SymbolIndex::KIND_LABELS[$kind])),
156            );
157        }
158
159        return $html;
160    }

162    /**
163     * Renders the namespace cell of one row, linking to its listing.
164     */
165    public function namespaceCell(RenderKit $services, string $pagePath, SymbolRow $row): string
166    {
167        $packageName = $this->packageOf($services, $row);
168        if ($row->namespace === '' || $packageName === '') {
169            return sprintf('<td class="item-ns">%s</td>', $services->escaper->e($row->namespace));
170        }
171
172        return sprintf(
173            '<td class="item-ns"><a href="%s">%s</a></td>',
174            $services->escaper->e($services->url->href($pagePath, $services->url->namespacePage($packageName, $row->namespace))),
175            $services->escaper->e($row->namespace),
176        );
177    }

179    /**
180     * Returns the package name that owns one symbol row.
181     */
182    public function packageOf(RenderKit $services, SymbolRow $row): string
183    {
184        $classLike = $services->model->symbolTable->classLike($row->fqcn);
185        if ($classLike !== null) {
186            return $classLike->packageName;
187        }
188
189        $function = $services->model->symbolTable->functionNamed($row->fqcn);
190
191        return $function !== null ? $function->packageName : '';
192    }

211    /**
212     * Renders the sidebar section anchors of a kind-grouped listing.
213     *
214     * @param list<SymbolRow> $rows
215     *
216     * @return list<array{id: string, label: string, status: string}>
217     */
218    public function sections(array $rows): array
219    {
220        $sections = [];
221        foreach ($this->symbols->byKind($rows) as $kind => $kindRows) {
222            $sections[] = [
223                'id' => SymbolIndex::KIND_ANCHORS[$kind] ?? $kind,
224                'label' => SymbolIndex::KIND_LABELS[$kind],
225                'status' => (new DiffStatus())->combine($this->statuses($kindRows)),
226            ];
227        }
228
229        return $sections;
