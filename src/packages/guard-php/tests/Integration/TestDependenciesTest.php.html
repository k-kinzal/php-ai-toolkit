212    private function remove(string $root): void
213    {
214        foreach (new RecursiveIteratorIterator(new RecursiveDirectoryIterator($root, FilesystemIterator::SKIP_DOTS), RecursiveIteratorIterator::CHILD_FIRST) as $file) {
215            if ($file->isDir()) {
216                rmdir($file->getPathname());
217            } else {
218                unlink($file->getPathname());
219            }
220        }
221        rmdir($root);
