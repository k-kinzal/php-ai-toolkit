18/**
19 * Diffs two analyzed revisions into one model the site is rendered from.
20 *
21 * The result is the head revision plus everything the head dropped, with
22 * every element marked in the diff index. Rendering one model instead of
23 * two sites is what lets a reader switch between the plain documentation
24 * and the diff without leaving the page.
25 */
26final class ProjectDiffer
27{
28    /** @readonly */
29    private ClassLikeMerger $classLikeMerger;
30
31    /** @readonly */
32    private FunctionMerger $functionMerger;
33
34    /** @readonly */
35    private DocumentDiffer $documents;
36
37    /** @readonly */
38    private DiffStatus $status;
39
40    /**
41     * Creates a project differ from its merging collaborators.
42     */
43    public function __construct(
44        ?ClassLikeMerger $classLikeMerger = null,
45        ?FunctionMerger $functionMerger = null,
46        ?DocumentDiffer $documents = null,
47        ?DiffStatus $status = null,
48    ) {
49        $this->classLikeMerger = $classLikeMerger ?? new ClassLikeMerger();
50        $this->functionMerger = $functionMerger ?? new FunctionMerger();
51        $this->documents = $documents ?? new DocumentDiffer();
52        $this->status = $status ?? new DiffStatus();
53    }
54
55    /**
56     * Diffs two project models into the merged model of the site.
57     */
58    public function diff(ProjectModel $base, ProjectModel $head, DiffIndex $index): ProjectModel
59    {
60        $classLikes = $this->classLikes($base, $head, $index);
61        $functions = $this->functions($base, $head, $index);
62        $packages = $this->packages($base, $head);
63        $symbolTable = new SymbolTable();
64        foreach ($classLikes as $classLike) {
65            $symbolTable->registerClassLike($classLike);
66        }
67
68        foreach ($functions as $function) {
69            $symbolTable->registerFunction($function);
70        }
71
72        $hierarchy = new HierarchyIndex();
73        $hierarchy->build($classLikes);
74        $this->markScopes($classLikes, $functions, $index);
75
76        return new ProjectModel(
77            $head->title,
78            $head->root,
79            $packages,
80            $head->graph,
81            $classLikes,
82            $functions,
83            $symbolTable,
84            $hierarchy,
85            $head->usages,
86            $head->testCases,
87            $head->layers,
88            array_merge($base->layerAssignments, $head->layerAssignments),
89            $head->coverage,
90            $head->warnings,
91            $this->documents->merge($base, $head, $index),
92        );
93    }

95    /**
96     * Merges the class-like symbols of both revisions.
97     *
98     * @return list<ClassLikeDoc>
99     */
100    public function classLikes(ProjectModel $base, ProjectModel $head, DiffIndex $index): array
101    {
102        $baseByName = [];
103        foreach ($base->classLikes as $classLike) {
104            $baseByName[strtolower($classLike->fqcn)] = $classLike;
105        }
106
107        $merged = [];
108        $seen = [];
109        foreach ($head->classLikes as $classLike) {
110            $key = strtolower($classLike->fqcn);
111            $seen[$key] = true;
112            $counterpart = $baseByName[$key] ?? null;
113            $merged[] = $counterpart === null
114                ? $this->classLikeMerger->single($classLike, DiffStatus::ADDED, $index)
115                : $this->classLikeMerger->merge($counterpart, $classLike, $index);
116        }
117
118        foreach ($baseByName as $key => $classLike) {
119            if (!isset($seen[$key])) {
120                $merged[] = $this->classLikeMerger->single($classLike, DiffStatus::REMOVED, $index);
121            }
122        }
123
124        return $merged;
125    }

127    /**
128     * Merges the top-level functions of both revisions.
129     *
130     * @return list<FunctionDoc>
131     */
132    public function functions(ProjectModel $base, ProjectModel $head, DiffIndex $index): array
133    {
134        $baseByName = [];
135        foreach ($base->functions as $function) {
136            $baseByName[strtolower($function->fqn)] = $function;
137        }
138
139        $merged = [];
140        $seen = [];
141        foreach ($head->functions as $function) {
142            $key = strtolower($function->fqn);
143            $seen[$key] = true;
144            $counterpart = $baseByName[$key] ?? null;
145            $merged[] = $counterpart === null
146                ? $this->functionMerger->single($function, DiffStatus::ADDED, $index)
147                : $this->functionMerger->merge($counterpart, $function, $index);
148        }
149
150        foreach ($baseByName as $key => $function) {
151            if (!isset($seen[$key])) {
152                $merged[] = $this->functionMerger->single($function, DiffStatus::REMOVED, $index);
153            }
154        }
155
156        return $merged;
157    }

159    /**
160     * Merges the documented packages of both revisions.
161     *
162     * @return list<DiscoveredPackage>
163     */
164    public function packages(ProjectModel $base, ProjectModel $head): array
165    {
166        $packages = [];
167        $seen = [];
168        foreach ($head->packages as $package) {
169            $seen[$package->manifest->name] = true;
170            $packages[] = $package;
171        }
172
173        foreach ($base->packages as $package) {
174            if (!isset($seen[$package->manifest->name])) {
175                $packages[] = $package;
176            }
177        }
178
179        return $packages;
180    }

182    /**
183     * Records the state of every package and namespace of the site.
184     *
185     * A scope is as changed as the symbols it holds, which is what lets the
186     * navigation hide the parts of the tree a revision never touched.
187     *
188     * @param list<ClassLikeDoc> $classLikes
189     * @param list<FunctionDoc> $functions
190     */
191    public function markScopes(array $classLikes, array $functions, DiffIndex $index): void
192    {
193        $keys = $index->keys();
194        $scopes = [];
195        foreach ($classLikes as $classLike) {
196            $status = $index->status($keys->classLike($classLike->fqcn));
197            $scopes[$keys->package($classLike->packageName)][] = $status;
198            $scopes[$keys->namespaceName($classLike->packageName, $classLike->namespace)][] = $status;
199        }
200
201        foreach ($functions as $function) {
202            $status = $index->status($keys->functionSymbol($function->fqn));
203            $scopes[$keys->package($function->packageName)][] = $status;
204            $scopes[$keys->namespaceName($function->packageName, $function->namespace)][] = $status;
205        }
206
207        foreach ($scopes as $key => $statuses) {
208            $index->mark($key, $this->status->combine($statuses));
209        }
210    }
