142    /**
143     * Renders the whole site into the output directory.
144     *
145     * @param ?DiffIndex $diff the comparison the site displays, if any
146     * @param ?int $workers how many workers to use, or null for the default
147     * @param ?RenderCache $cache what the previous run wrote into this directory
148     *
149     * @return int the number of pages the site holds
150     *
151     * @throws DocGenException when a page cannot be rendered or written
152     */
153    public function render(ProjectModel $model, string $outputRoot, ?DiffIndex $diff = null, ?int $workers = null, ?RenderCache $cache = null): int
154    {
155        $services = $this->services($model, $diff);
156        $writer = new CachedPageWriter($this->writer, $cache);
157        $this->assets->publish($outputRoot);
158        $this->assets->publishCard($outputRoot, $model->baseUrl, $model->title, $this->indexPage->description($services));
159        $this->writer->write($outputRoot, 'assets/search-index.js', $this->searchIndex->build($model, $services->diff));
160        $records = [$writer->write(
161            $outputRoot,
162            'index.html',
163            $this->signatures->index($services),
164            fn (): string => $this->indexPage->render($services),
165        )];
166        foreach ([
167            $this->renderPackagePages($services, $model, $outputRoot, $writer),
168            $this->renderClassLikePages($services, $model, $outputRoot, $writer, $workers),
169            $this->renderFunctionPages($services, $model, $outputRoot, $writer),
170            $this->renderDocumentPages($services, $model, $outputRoot, $writer),
171            $this->renderSourcePages($services, $model, $outputRoot, $writer, $workers),
172        ] as $group) {
173            foreach ($group as $record) {
174                $records[] = $record;
175            }
176        }
177
178        $cache?->record($outputRoot, $records);
179
180        return count($records);
181    }

183    /**
184     * Renders one page per documented class-like symbol.
185     *
186     * This is the largest and the most expensive group of pages a site has,
187     * so it is the group that is split across workers. Each page is written
188     * by exactly one process and depends on nothing but the finished model,
189     * so the site does not depend on how the work was divided.
190     *
191     * @param ?int $workers how many workers to use, or null for the default
192     *
193     * @return list<PageRecord>
194     *
195     * @throws DocGenException when a worker cannot finish its pages
196     */
197    public function renderClassLikePages(RenderKit $services, ProjectModel $model, string $outputRoot, CachedPageWriter $writer, ?int $workers = null): array
198    {
199        $documented = [];
200        foreach ($model->classLikes as $classLike) {
201            if (!$classLike->isDev) {
202                $documented[] = $classLike;
203            }
204        }
205
206        return $writer->records($this->workers->map(
207            $this->scheduler->plan($documented, static fn (ClassLikeDoc $classLike): int => $classLike->endLine - $classLike->startLine, $workers),
208            function (array $job) use ($services, $outputRoot, $writer): array {
209                $records = [];
210                foreach ($job as $classLike) {
211                    $records[] = $writer->write(
212                        $outputRoot,
213                        $this->url->classLikePage($classLike),
214                        $this->signatures->classLike($services, $classLike),
215                        fn (): string => $this->classLikePage->render($services, $classLike),
216                    );
217                }
218
219                return $records;
220            },
221        ));
222    }

248    /**
249     * Renders one highlighted page per source file of the model.
250     *
251     * A file that only the base revision had is rendered from the base
252     * checkout, so the source a removed symbol links to stays readable.
253     *
254     * Highlighting every line of every documented file is the second most
255     * expensive group of pages, so it is split across workers as well.
256     *
257     * @param ?int $workers how many workers to use, or null for the default
258     *
259     * @return list<PageRecord>
260     *
261     * @throws DocGenException when a worker cannot finish its pages
262     */
263    public function renderSourcePages(RenderKit $services, ProjectModel $model, string $outputRoot, CachedPageWriter $writer, ?int $workers = null): array
264    {
265        $files = $this->pages->sourceFiles($model);
266        $root = $model->root;
267
268        return $writer->records($this->workers->map(
269            $this->scheduler->plan($files, static fn (string $file): int => (int) @filesize($root . '/' . $file), $workers),
270            fn (array $job): array => $this->writeSourcePages($services, $root, $outputRoot, $writer, $job),
271        ));
272    }

274    /**
275     * Writes the source pages of one job and reports what it wrote.
276     *
277     * @param list<string> $files project-relative source files
278     *
279     * @return list<PageRecord>
280     *
281     * @throws DocGenException when a page cannot be written
282     */
283    public function writeSourcePages(RenderKit $services, string $root, string $outputRoot, CachedPageWriter $writer, array $files): array
284    {
285        $records = [];
286        foreach ($files as $relativeFile) {
287            $code = $this->pages->contents($root . '/' . $relativeFile);
288            $baseCode = $services->diff->baseSource($relativeFile);
289            if ($code === null && $baseCode === null) {
290                continue;
291            }
292
293            $records[] = $writer->write(
294                $outputRoot,
295                $this->url->sourcePage($relativeFile),
296                $this->signatures->source($services, $relativeFile, $code, $baseCode),
297                fn (): string => $this->sourcePage->render($services, $relativeFile, $code, $baseCode),
298            );
299        }
300
301        return $records;
302    }

305     * Renders the package and namespace pages of every package.
306     *
307     * @return list<PageRecord>
308     *
309     * @throws DocGenException when a page cannot be written
310     */
311    public function renderPackagePages(RenderKit $services, ProjectModel $model, string $outputRoot, CachedPageWriter $writer): array
312    {
313        $records = [];
314        foreach ($model->packages as $package) {
315            $name = $package->manifest->name;
316            $readme = $this->pages->readme($package);
317            $records[] = $writer->write(
318                $outputRoot,
319                $this->url->packagePage($name),
320                $this->signatures->package($services, $package, $readme),
321                fn (): string => $this->packagePage->render($services, $package, $readme),
322            );
323            $records[] = $writer->write(
324                $outputRoot,
325                $this->url->allItemsPage($name),
326                $this->signatures->allItems($services, $name),
327                fn (): string => $this->allItemsPage->render($services, $name),
328            );
329            foreach ($this->sidebar->packageLayers($services, $name) as $layer) {
330                $records[] = $writer->write(
331                    $outputRoot,
332                    $this->url->layerPage($name, $layer),
333                    $this->signatures->layer($services, $name, $layer),
334                    fn (): string => $this->layerPage->render($services, $name, $layer),
335                );
336            }
337
338            foreach ($this->namespacePages($services, $model, $outputRoot, $writer, $name) as $record) {
339                $records[] = $record;
340            }
341        }
342
343        return $records;
344    }

373     * Renders one page per Markdown document of the project.
374     *
375     * @return list<PageRecord>
376     *
377     * @throws DocGenException when a page cannot be written
378     */
379    public function renderDocumentPages(RenderKit $services, ProjectModel $model, string $outputRoot, CachedPageWriter $writer): array
380    {
381        $records = [];
382        foreach ($model->documents as $document) {
383            $markdown = $this->pages->contents($model->root . '/' . $document->file);
384            $baseMarkdown = $services->diff->baseSource($document->file);
385            if ($markdown === null && $baseMarkdown === null) {
386                continue;
387            }
388
389            $records[] = $writer->write(
390                $outputRoot,
391                $this->url->documentPage($document->packageName, $document->path),
392                $this->signatures->document($services, $document, $markdown, $baseMarkdown),
393                fn (): string => $this->documentPage->render($services, $document, $markdown, $baseMarkdown),
394            );
395        }
396
397        return $records;
398    }
