11/**
12 * Detects whether a PHPDoc block documents an example doctest can run.
13 *
14 * Extraction is delegated to the doctest grammar rather than repeated here, so
15 * the rule requires exactly what the doctest test suite would execute: a block
16 * this rule accepts is a block that becomes a PHPUnit test case.
17 */
18final class RunnableExampleDetector
19{
20    /** @readonly */
21    private ExampleExtractor $extractor;
22
23    /**
24     * Creates a detector from the doctest example grammar.
25     */
26    public function __construct(?ExampleExtractor $extractor = null)
27    {
28        $this->extractor = $extractor ?? new ExampleExtractor();
29    }
30
31    /**
32     * Reports whether the PHPDoc block carries at least one runnable example.
33     *
34     * A single-line at-example tag carries a description and no code, so the
35     * extractor yields nothing for it and it does not satisfy the requirement.
36     */
37    public function hasRunnableExample(?string $docComment): bool
38    {
39        if ($docComment === null) {
40            return false;
41        }
42
43        $target = new Target(TargetKind::CLASS_LIKE, '', $docComment, '', 0);
44        foreach ($this->extractor->extract($target) as $example) {
45            return true;
46        }
47
48        return false;
49    }
50}
