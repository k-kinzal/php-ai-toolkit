72    /**
73     * Renders one method section with docs, guarantees, and call sites.
74     */
75    public function method(RenderKit $services, string $pagePath, ClassLikeDoc $owner, MethodDoc $method, TypeRenderContext $context): string
76    {
77        $anchor = 'method.' . $method->name;
78        $key = $services->diff->methodKey($owner->fqcn, $method->name);
79        $html = sprintf('<div class="member"%s id="%s">', $services->diff->attribute($key), $services->escaper->e($anchor));
80        $html .= '<div class="member-head">';
81        $html .= '<pre class="member-sig"><code>' . $this->signature->methodSignature($services, $method, $context, $key) . '</code></pre>';
82        $html .= $this->meta($services, $pagePath, $owner->file, $method->startLine, $method->endLine, $anchor);
83        $html .= '</div>';
84        $html .= '<div class="member-body">';
85        $html .= $this->docText->render($services, $method->docBlock, $context, sprintf('%s::%s()', $owner->shortName, $method->name));
86        $html .= $this->paramTable($services, $method, $context, $key);
87        $html .= $this->tagExamples($services, $method->docBlock, sprintf('%s::%s()', $owner->shortName, $method->name));
88        if (!$services->model->publicApi) {
89            $html .= $this->testCases->section($services, $pagePath, $services->model->testCases->forMember($owner->fqcn, $method->name));
90            $html .= $this->usageList->section($services, $pagePath, 'Called from', $this->callers($services, $owner, $method), false);
91            $html .= $this->usageList->callSection($services, $pagePath, 'Calls', $services->model->usages->callsFrom($owner->fqcn, $method->name));
92        }
93
94        return $html . '</div></div>' . "\n";
95    }

97    /**
98     * Collects the production call sites of one method.
99     *
100     * @return list<Usage>
101     */
102    public function callers(RenderKit $services, ClassLikeDoc $owner, MethodDoc $method): array
103    {
104        $callers = [];
105        foreach ($services->model->usages->forMember($owner->fqcn, $method->name, false) as $usage) {
106            if (in_array($usage->kind, self::CALL_KINDS, true)) {
107                $callers[] = $usage;
108            }
109        }
110
111        return $callers;
112    }

162    /**
163     * Renders the coverage figure, source link, and anchor of a member.
164     *
165     * The coverage chip is omitted when no coverage report was loaded, so a
166     * missing report never reads as untested code.
167     */
168    public function meta(RenderKit $services, string $pagePath, string $file, int $startLine, int $endLine, string $anchor): string
169    {
170        $html = '<div class="member-meta">';
171        $coverage = $services->model->coverage;
172        if ($coverage !== null) {
173            $method = $coverage->methodAt($file, $startLine, $endLine);
174            if ($method !== null) {
175                $level = $method->percent >= 90.0 ? 'high' : ($method->percent >= 50.0 ? 'mid' : 'low');
176                $html .= sprintf(
177                    '<span class="chip chip-sm chip-cov-%s" title="%d of %d executable lines executed by the test suite">%.0f%%</span>',
178                    $level,
179                    $method->executed,
180                    $method->executable,
181                    $method->percent,
182                );
183            }
184        }
185
186        $html .= sprintf(
187            '<a class="src-link" href="%s">source</a><a class="anchor" href="#%s">§</a>',
188            $services->escaper->e($services->url->href($pagePath, $services->url->sourcePage($file)) . '#L' . $startLine),
189            $services->escaper->e($anchor),
190        );
191
192        return $html . '</div>';
193    }

209     * Renders the parameters, return, and throws of a callable declaration.
210     *
211     * Each of the three carries a different question, so each gets its own
212     * labeled block instead of one merged table. Functions and methods
213     * share the rendering, so both read the same way.
214     *

219    {
220        return $this->parameterSection($services, $parameters, $context, $ownerKey)
221            . $this->mutationSection($services, $parameters, $docBlock)

223            . $this->throwsSection($services, $docBlock, $context, $ownerKey);

226    /**
227     * Renders the parameter block of a callable declaration.
228     *
229     * @param list<ParameterDoc> $parameters
230     * @param string $ownerKey the diff key the parameter states are under
231     */
232    public function parameterSection(RenderKit $services, array $parameters, TypeRenderContext $context, string $ownerKey = ''): string
233    {
234        if ($parameters === []) {
235            return '';
236        }
237
238        $rows = '';
239        $statuses = [];
240        foreach ($parameters as $parameter) {
241            $annotated = $parameter->type->annotated;
242            $statuses[] = $ownerKey === '' ? DiffStatus::SAME : $services->diff->parameterStatus($ownerKey, $parameter->name);
243            $rows .= sprintf(
244                '<tr%s><td><code class="t-var">$%s</code>%s</td><td><code>%s</code></td><td>%s</td></tr>',
245                $ownerKey === '' ? '' : $services->diff->parameter($ownerKey, $parameter->name),
246                $services->escaper->e($parameter->name),
247                $parameter->mutable ? ' <span class="chip chip-sm">mut</span>' : '',
248                $services->typeHtml->render($annotated !== null ? $annotated->type : null, $parameter->type->native, $context),
249                $services->escaper->e($parameter->description),
250            );
251        }
252
253        return $this->block(
254            'Parameters',
255            '<div class="table-wrap"><table class="param-table">' . $rows . '</table></div>',
256            $ownerKey === '' ? '' : $services->diff->combined($statuses),
257        );
258    }

260    /**
261     * Renders the externally visible effects declared by a callable.
262     *
263     * @param list<ParameterDoc> $parameters
264     */
265    public function mutationSection(RenderKit $services, array $parameters, ?DocBlock $docBlock): string
266    {
267        if ($docBlock === null) {
268            return '';
269        }
270
271        $targets = [];
272        foreach ($parameters as $parameter) {
273            if ($parameter->mutable) {
274                $targets[] = '$' . $parameter->name;
275            }
276        }
277
278        if ($docBlock->mutation->mutatesThis()) {
279            $targets[] = '$this';
280        }
281
282        if ($docBlock->mutation->mutatesGlobal()) {
283            $targets[] = 'global';
284        }
285
286        if ($targets === []) {
287            return '';
288        }
289
290        $items = '';
291        foreach ($targets as $target) {
292            $items .= '<code>' . $services->escaper->e($target) . '</code> ';
293        }
294
295        return $this->block('Mutates', '<div class="type-row">' . trim($items) . '</div>');
296    }

298    /**
299     * Renders the return block, or nothing for a void declaration.
300     *
301     * @param string $ownerKey the diff key of the declaration
302     */
303    public function returnSection(RenderKit $services, TypeSignature $returnType, TypeRenderContext $context, string $ownerKey = ''): string
304    {
305        $annotated = $returnType->annotated;
306        $type = $services->typeHtml->render($annotated !== null ? $annotated->type : null, $returnType->native, $context);
307        if ($type === '' || $returnType->native === 'void') {
308            return '';
309        }
310
311        $description = $annotated !== null ? $annotated->description : '';
312
313        return $this->block('Returns', sprintf(
314            '<div class="type-row"><code>%s</code>%s</div>',
315            $type,
316            $description !== '' ? ' <span class="type-note">' . $services->escaper->e($description) . '</span>' : '',
317        ), $ownerKey === '' ? '' : $services->diff->returnType($ownerKey));
318    }

320    /**
321     * Renders the throws block of a member.
322     *
323     * @param string $ownerKey the diff key of the declaration
324     */
325    public function throwsSection(RenderKit $services, ?DocBlock $docBlock, TypeRenderContext $context, string $ownerKey = ''): string
326    {
327        if ($docBlock === null || $docBlock->throws === []) {
328            return '';
329        }
330
331        $rows = '';
332        foreach ($docBlock->throws as $tag) {
333            $rows .= sprintf(
334                '<div class="type-row"><code>%s</code>%s</div>',
335                $tag->type !== null ? $services->typeHtml->node($tag->type, $context) : '',
336                $tag->description !== '' ? ' <span class="type-note">' . $services->escaper->e($tag->description) . '</span>' : '',
337            );
338        }
339
340        return $this->block('Throws', $rows, $ownerKey === '' ? '' : $services->diff->throwsTags($ownerKey));
341    }

354     * Renders the at-example doctests of a member docblock.
355     *
356     * @param string $symbol the unqualified target name doctest names the examples after, empty when unknown
357     */
358    public function tagExamples(RenderKit $services, ?DocBlock $docBlock, string $symbol = ''): string
359    {
360        if ($docBlock === null) {
361            return '';
362        }
363
364        $html = '';
365        foreach ($services->doctest->extract($docBlock->raw) as $example) {
366            if ($example->source !== 'fence') {
367                $name = $symbol === '' ? '' : $this->example->exampleName($symbol, $example->description, $example->index);
368                $html .= $this->example->figure($services, $example->description, $example->code, $example->source === 'tag', $name);
369            }
370        }
371
372        return $html;
373    }
