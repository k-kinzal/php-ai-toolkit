30    public function testReadParsesTheModernWrappedForm(): void
31    {
32        $dir = sys_get_temp_dir() . '/docgen-layer-' . uniqid('', true);
33        mkdir($dir, 0777, true);
34        file_put_contents($dir . '/deptrac.yaml', <<<'YAML'
35deptrac:
36  layers:
37    - name: Domain
38      collectors:
39        - type: directory
40          value: 'src/Domain/.*'
41    - name: Application
42      collectors: ~
43  ruleset:
44    Domain: ~
45    Application:
46      - Domain
47YAML);
48
49        $model = (new DeptracConfigReader())->read($dir . '/deptrac.yaml');
50
51        self::assertCount(2, $model->layers);
52        self::assertSame('Domain', $model->layers[0]->name);
53        self::assertCount(1, $model->layers[0]->collectors);
54        self::assertSame('directory', $model->layers[0]->collectors[0]->type);
55        self::assertSame('src/Domain/.*', $model->layers[0]->collectors[0]->value);
56        self::assertSame('Application', $model->layers[1]->name);
57        self::assertSame([], $model->layers[1]->collectors);
58        self::assertSame(['Domain' => [], 'Application' => ['Domain']], $model->ruleset);
59    }
