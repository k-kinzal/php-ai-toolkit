190    public function relative(string $root, string $target): string
191    {
192        $from = $this->parts($this->absolute($root));
193        $to = $this->parts($this->absolute($target));
194        while ($from !== [] && $to !== [] && $from[0] === $to[0]) {
195            array_shift($from);
196            array_shift($to);
197        }
198        $prefix = $from === [] ? '' : str_repeat('../', count($from));
199
200        return $prefix . implode('/', $to);
201    }

223    /**
224     * Reads one shipped preset.
225     *
226     * @return array<string, mixed>
227     * @throws PolicyException when the preset is missing or not a mapping
228     */
229    public function read(string $name): array
230    {
231        $path = $this->file($name);
232        if (!is_file($path)) {
233            throw new PolicyException('Guard preset "' . $name . '" was not found at ' . $path . '. Install k-kinzal/guard-php so its rules/ directory is present.');
234        }
235        try {
236            $parsed = Yaml::parseFile($path);
237        } catch (ParseException $exception) {
238            throw new PolicyException('Invalid guard preset ' . $name . ': ' . $exception->getMessage() . '. Correct rules/' . $name . '.yaml.', 0, $exception);
239        }
240        if (!is_array($parsed) || ($parsed !== [] && array_values($parsed) === $parsed)) {
241            throw new PolicyException('Guard preset "' . $name . '" must be a mapping. Correct rules/' . $name . '.yaml.');
242        }
243        $document = [];
244        foreach ($parsed as $key => $value) {
245            if (is_string($key)) {
246                $document[$key] = $value;
247            }
248        }
249
250        return $document;
251    }

254     * Returns the configuration rule ids declared by one preset.
255     *
256     * @return list<string>
257     * @throws PolicyException when the preset has no configuration rule ids
258     */
259    public function ruleIds(string $name): array
260    {
261        $configuration = $this->read($name)['configuration'] ?? null;
262        if (!is_array($configuration) || array_values($configuration) !== $configuration) {
263            throw new PolicyException('rules/' . $name . '.yaml configuration must be a list of field rules.');
264        }
265        $ids = [];
266        foreach ($configuration as $rule) {
267            if (!is_array($rule) || !is_string($rule['id'] ?? null) || $rule['id'] === '') {
268                throw new PolicyException('rules/' . $name . '.yaml configuration rules must have a non-empty id.');
269            }
270            $ids[] = $rule['id'];
271        }
272
273        return $ids;
274    }

277     * Returns the standard metric limits shipped in metrics.yaml.
278     *
279     * @return array<string, array<string, int>>
280     * @throws PolicyException when metrics.yaml does not define integer limits
281     */
282    public function limits(): array
283    {
284        $metrics = $this->read('metrics')['metrics'] ?? null;
285        $profiles = is_array($metrics) ? ($metrics['profiles'] ?? null) : null;
286        $standard = is_array($profiles) ? ($profiles['standard'] ?? null) : null;
287        $limits = is_array($standard) ? ($standard['limits'] ?? null) : null;
288        if (!is_array($limits) || !$this->isMapping($limits)) {
289            throw new PolicyException('rules/metrics.yaml must define metrics.profiles.standard.limits.');
290        }
291
292        return $this->metrics($limits);
293    }

295    /**
296     * Checks that every limit kind maps to integer metrics.
297     *
298     * @param array<mixed> $limits
299     * @return array<string, array<string, int>>
300     * @throws PolicyException when a limit is not an integer
301     */
302    public function metrics(array $limits): array
303    {
304        $result = [];
305        foreach ($limits as $kind => $metrics) {
306            if (!is_string($kind) || !is_array($metrics) || !$this->isMapping($metrics)) {
307                throw new PolicyException('rules/metrics.yaml limits must map each kind to numeric metrics.');
308            }
309            $values = [];
310            foreach ($metrics as $metric => $value) {
311                if (!is_string($metric) || !is_int($value)) {
312                    $label = is_string($metric) ? $kind . '.' . $metric : $kind;
313                    throw new PolicyException('rules/metrics.yaml limit "' . $label . '" must be an integer.');
314                }
315                $values[$metric] = $value;
316            }
317            $result[$kind] = $values;
318        }
319
320        return $result;
321    }
