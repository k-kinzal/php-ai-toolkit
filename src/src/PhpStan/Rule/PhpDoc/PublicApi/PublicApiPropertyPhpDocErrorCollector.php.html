13final class PublicApiPropertyPhpDocErrorCollector

15    /**

17     */
18    public function errors(\PhpParser\Node\Stmt\ClassLike $node, string $className): array
19    {
20        $errors = [];
21        foreach ($node->getProperties() as $property) {
22            if (!$property->isPublic()) {
23                continue;
24            }
25
26            $names = [];
27            foreach ($property->props as $declared) {
28                $names[] = '$' . $declared->name->toString();
29            }
30
31            if ($property->getDocComment() === null) {
32                $errors[] = RuleErrorBuilder::message(
33                    sprintf(
34                        'Add a multi-line PHPDoc block to public property %s::%s describing the property.',
35                        $className,
36                        implode(', ', $names)
37                    )
38                )
39                    ->identifier('customRules.requirePhpDocOnProperty')
40                    ->line($property->getStartLine())
41                    ->build();
42            }
43        }
44
45        return $errors;
46    }
