18/**
19 * Base class behind the toolkit's PHPUnit doctest suite.
20 *
21 * A project on PHPUnit 10 or later normally does not extend this class. It
22 * points its PHPUnit configuration at the package's DoctestSuite.php and
23 * supplies scan paths to DoctestExtension. Extending this runner is a custom
24 * integration point, not the setup path. Each example found becomes one
25 * PHPUnit test case.
26 *
27 * This class binds its data provider with a PHPUnit attribute and therefore
28 * needs PHPUnit 10 or later. On PHPUnit 9 extend
29 * Toolkit\Doctest\TestCase\Legacy\LegacyDoctestRunner, which binds the
30 * same provider with a doc-comment annotation.
31 *
32 * An example exercises whatever code it documents, so a suite declares no
33 * coverage target of its own: mark the subclass #[CoversNothing], the way
34 * DoctestSuite is. The attribute has to sit on the concrete class, because
35 * PHPUnit reads class metadata from the class it runs rather than from what
36 * that class extends, and PHPUnit 12 deprecated carrying it on the test method.
37 *
38 * This is the only entry point the port carries. k-kinzal/doctest-php also
39 * ships a DoctestCase wrapping one example, which nothing there uses and which
40 * cannot work here: it names itself through the PHPUnit constructor, and
41 * PHPUnit 10 made TestCase::__construct() final.
42 */
43abstract class DoctestRunner extends TestCase
44{
45    private static ?ExampleExecutor $executor = null;
46
47    /**
48     * Override to provide custom configuration.
49     *
50     * @return Configuration the configuration for doctest scanning
51     */
52    abstract public static function configure(): Configuration;
53
54    /**
55     * Provides examples as test data.
56     *
57     * @return Generator<string, array{?Example}>
58     */
59    public static function doctestProvider(): Generator
60    {
61        $config = static::configure();
62        $fileScanner = new FileScanner($config);
63        $sourceScanner = new SourceScanner();
64        $extractor = new ExampleExtractor();
65        $found = false;
66
67        foreach ($fileScanner->scan() as $filePath) {
68            foreach ($sourceScanner->scanFile($filePath) as $target) {
69                foreach ($extractor->extract($target) as $example) {
70                    $found = true;
71                    yield $example->getName() => [$example];
72                }
73            }
74        }
75
76        if (!$found) {
77            yield 'No doctest examples found' => [null];
78        }
79    }
80
81    /**
82     * Tests a docblock example.
83     *
84     * @param ?Example $example the example to test, or null when none were discovered
85     */
86    #[Test]
87    #[DataProvider('doctestProvider')]
88    public function testDocblockExample(?Example $example): void
89    {
90        if ($example === null) {
91            $this->addToAssertionCount(1);
92
93            return;
94        }
95
96        if (self::$executor === null) {
97            self::$executor = new ExampleExecutor(static::configure()->getBootstrap());
98        }
99
100        $result = self::$executor->execute($example);
101
102        self::assertTrue($result->passed, $result->getErrorMessage());
103    }
104}
