37    public function create(string $class, array $options): Extension
38    {
39        if (!class_exists($class)) {
40            throw new PolicyException('Class "' . $class . '" was not found. Install it or add it to Composer autoload and run composer dump-autoload.');
41        }
42        if (!is_a($class, Extension::class, true)) {
43            throw new PolicyException('Class "' . $class . '" must implement ' . Extension::class . '.');
44        }
45        $reflection = new ReflectionClass($class);
46        if ($reflection->isAbstract()) {
47            throw new PolicyException('Class "' . $class . '" is abstract. Configure a concrete extension class.');
48        }
49        if (is_a($class, ConfigurableExtension::class, true)) {
50            return $class::fromOptions($options);
51        }
52        if ($options !== []) {

54        }
55        $constructor = $reflection->getConstructor();
