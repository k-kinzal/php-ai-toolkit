43    /**
44     * Returns a diagnostic description for rule listings, independent of current violations.
45     */
46    public function forRule(string $id): string
47    {
48        if (str_starts_with($id, 'metrics.')) {
49            return $this->diagnostic($id, str_contains($id, 'complexity')
50                ? 'The function or method exceeds its configured cyclomatic complexity limit. Simplify conditional branches and extract independent operations into focused methods until each stays within the limit.'
51                : 'The source unit exceeds its configured line limit. Split unrelated responsibilities into smaller files, classes or methods until each stays within the limit; preserve behavior and tests.');
52        }
53        return $this->diagnostic($id, self::MESSAGES[$id] ?? 'This custom rule needs a diagnostic message. Describe the offending input and the concrete edit that resolves it in the rule definition.');
54    }
