28     * Posts every payload to a path and returns the decoded responses in the same order.
29     *
30     * @param non-empty-string $path the endpoint path, starting with a slash
31     * @param list<array<string, mixed>> $payloads
32     * @return list<array<string, mixed>>
33     * @throws LintException when a request fails or a response is not a JSON object
34     */
35    public function post(string $path, array $payloads): array
36    {
37        $requests = [];
38        foreach ($payloads as $payload) {
39            $requests[] = $this->request($path, $payload);
40        }
41        $documents = [];
42        foreach ($this->transport->send($requests) as $index => $response) {
43            $documents[] = $this->decode($response, $requests[$index]);
44        }

49    /**
50     * Builds the request for one payload.
51     *
52     * @param non-empty-string $path the endpoint path, starting with a slash
53     * @param array<string, mixed> $payload
54     * @throws LintException when the payload cannot be encoded
55     */
56    public function request(string $path, array $payload): HttpRequest
57    {
58        try {
59            $body = json_encode($payload, JSON_THROW_ON_ERROR | JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);
60        } catch (JsonException $exception) {
61            throw new LintException(sprintf('Could not encode a request for %s: %s. Make sure the linted files are valid UTF-8.', $this->settings->name, $exception->getMessage()), 0, $exception);
62        }
63
64        $headers = $this->apiKey === '' ? [] : ['Authorization' => 'Bearer ' . $this->apiKey];
65
66        return new HttpRequest('POST', rtrim($this->settings->baseUrl, '/') . $path, $headers + [
67            'Content-Type' => 'application/json',
68            'Accept' => 'application/json',
69            'User-Agent' => 'decisions-lint',
70        ], $body);
71    }

73    /**
74     * Decodes a successful response, or explains the failure.
75     *
76     * @return array<string, mixed>
77     * @throws LintException when the status is not 2xx or the body is not a JSON object
78     */
79    public function decode(HttpResponse $response, HttpRequest $request): array
80    {
81        $document = null;
82        try {
83            $document = json_decode($response->body, true, 64, JSON_THROW_ON_ERROR);
84        } catch (JsonException $exception) {
85            if ($response->ok()) {
86                throw new LintException(sprintf('%s returned HTTP %d for %s with a body that is not JSON: %s. Check provider.base_url in decisions.yaml.', $this->settings->name, $response->status, $request->url, $exception->getMessage()), 0, $exception);
87            }
88        }
89        if (!$response->ok()) {
90            throw new LintException($this->failure($response, $request, is_array($document) ? $document : []));
91        }
92        if (!is_array($document)) {
93            throw new LintException(sprintf('%s returned HTTP %d for %s with a JSON body that is not an object. Check provider.base_url in decisions.yaml.', $this->settings->name, $response->status, $request->url));
94        }
95        $object = [];
96        foreach ($document as $key => $value) {
97            $object[(string) $key] = $value;
98        }
99
100        return $object;
101    }

105     *

108    public function failure(HttpResponse $response, HttpRequest $request, array $document): string
109    {
110        $detail = $this->detail($document, $response->body);
111        $hint = match (true) {
112            $response->status === 401 || $response->status === 403 => sprintf('Check that the %s environment variable holds a valid %s API key.', $this->settings->apiKeyEnv, $this->settings->name),
113            $response->status === 404 => 'Check provider.base_url in decisions.yaml.',
114            $response->status === 422 || $response->status === 400 => sprintf('Check provider.model ("%s") and the rule questions in decisions.yaml.', $this->settings->model),
115            $response->status === 429 => 'Lower provider.concurrency in decisions.yaml or retry later.',
116            $response->status >= 500 => 'The provider is unavailable; retry later.',
117            default => 'Check the provider status and decisions.yaml.',
118        };
119
120        return sprintf('%s returned HTTP %d for POST %s%s. %s', $this->settings->name, $response->status, $request->url, $detail === '' ? '' : ': ' . $detail, $hint);
121    }

129    {
