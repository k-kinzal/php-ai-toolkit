185    /**
186     * @return array<string, array{ConstExprNode, string}>
187     *
188     * @throws RuntimeException when the installed parser reads no constant type
189     */
190    public static function providerConstantExpression(): array
191    {
192        $cases = [];
193        foreach (['string' => ["/** @param 'key' \$x */", "'key'"], 'integer' => ['/** @param 42 $x */', '42']] as $label => $case) {
194            $doc = (new DocBlockReader())->read($case[0]);
195            $type = $doc === null ? null : ($doc->params['$x']->type ?? null);
196            if (!$type instanceof ConstTypeNode) {
197                throw new RuntimeException('The installed parser read no constant type from the snippet.');
198            }
199
200            $cases[$label] = [$type->constExpr, $case[1]];
201        }
202
203        return $cases;
204    }
