11/**
12 * Loads a guard document and the documents it imports.
13 *
14 * Paths are relative to the file that lists them. Later imports override earlier ones, and the current file overrides its imports.
15 */
16final class ImportResolver
17{
18    /** @var list<string> */
19    private array $seen = [];
20
21    /** @var list<string> */
22    private array $paths = [];
23
24    /**
25     * Loads one guard document, following its imports.
26     *
27     * @return array<string, mixed>
28     * @throws PolicyException when the file is missing, circular, or not a version 1 document
29     */
30    public function resolve(string $path, bool $root = true): array
31    {
32        if (!is_file($path)) {
33            $message = $root
34                ? 'Guard configuration not found: ' . $path . '. Run guard init first.'
35                : 'Guard import not found: ' . $path . '. Check that the path is relative to the file that lists it.';
36            throw new PolicyException($message);
37        }
38        $identity = realpath($path);
39        $identity = $identity === false ? $path : $identity;
40        if (in_array($identity, $this->seen, true)) {
41            throw new PolicyException('Circular guard import: ' . implode(' -> ', array_merge($this->paths, [$path])) . '. Remove the repeated import.');
42        }
43        $this->seen[] = $identity;
44        $this->paths[] = $path;
45        try {
46            return $this->document($path, $root);
47        } finally {
48            array_pop($this->seen);
49            array_pop($this->paths);
50        }
51    }
52
53    /**
54     * Parses one document, loads its imports, and overlays the document on those imports.
55     *
56     * @return array<string, mixed>
57     * @throws PolicyException when the document is malformed
58     */
59    public function document(string $path, bool $root): array
60    {
61        $data = $this->mapping($path, $root);
62        $this->version($data, $path, $root);
63        $this->assertUnique($data, $path);
64        $merged = [];
65        foreach ($this->imports($data['imports'] ?? null, $path, $root) as $import) {
66            $merged = (new DocumentMerger())->merge($merged, $this->resolve($import, false));
67        }
68        unset($data['imports']);
69
70        return (new DocumentMerger())->merge($merged, $data);
71    }

73    /**
74     * Reads a YAML mapping and rejects unknown top-level keys.
75     *
76     * @return array<string, mixed>
77     * @throws PolicyException when the YAML cannot be parsed or contains unknown keys
78     */
79    public function mapping(string $path, bool $root): array
80    {
81        try {
82            $parsed = Yaml::parseFile($path);
83        } catch (ParseException $exception) {
84            $name = $root ? 'guard.yaml' : basename($path);
85            throw new PolicyException('Invalid ' . $name . ': ' . $exception->getMessage() . '. Correct the YAML syntax.', 0, $exception);
86        }
87        $context = $root ? 'guard.yaml' : basename($path);
88
89        return (new Schema())->mapping($parsed, ['version', 'imports', 'metrics', 'structure', 'documentation', 'configuration', 'extensions', 'collect'], $context);
90    }

92    /**
93     * Requires version 1 on the project file and on any imported file that sets version.
94     *
95     * @param array<string, mixed> $data
96     * @throws PolicyException when version is missing on the project file or is not 1
97     */
98    public function version(array $data, string $path, bool $root): void
99    {
100        if (!array_key_exists('version', $data)) {
101            if ($root) {
102                throw new PolicyException('guard.yaml.version must be 1. Use guard init to generate the current schema.');
103            }
104
105            return;
106        }
107        if ($data['version'] !== 1) {
108            $name = $root ? 'guard.yaml' : basename($path);
109            throw new PolicyException($name . '.version must be 1. Use guard init to generate the current schema.');
110        }
111    }

113    /**
114     * Rejects two configuration rules with the same id in one file.
115     *
116     * @param array<string, mixed> $data
117     * @throws PolicyException when a file repeats a configuration rule id
118     */
119    public function assertUnique(array $data, string $path): void
120    {
121        $rules = $data['configuration'] ?? null;
122        if (!is_array($rules) || array_values($rules) !== $rules) {
123            return;
124        }
125        $seen = [];
126        foreach ($rules as $rule) {
127            if (!is_array($rule) || !is_string($rule['id'] ?? null) || $rule['id'] === '') {
128                continue;
129            }
130            if (isset($seen[$rule['id']])) {
131                throw new PolicyException('Duplicate configuration rule "' . $rule['id'] . '" in ' . $path . '. Give each rule a unique id in that file.');
132            }
133            $seen[$rule['id']] = true;
134        }
135    }
136
137    /**
138     * Resolves import paths from the directory of the file that lists them.
139     *
140     * @param mixed $value
141     * @return list<string>
142     * @throws PolicyException when imports is not a list of paths
143     */
144    public function imports($value, string $path, bool $root): array
145    {
146        if ($value === null) {
147            return [];
148        }
149        $context = ($root ? 'guard.yaml' : basename($path)) . '.imports';
150        $directory = dirname($path);
151        $paths = [];
152        foreach ((new Schema())->strings($value, $context) as $import) {
153            $paths[] = str_starts_with($import, '/') ? $import : $directory . '/' . $import;
154        }
155
156        return $paths;
157    }
