137final class LegacyMigrationTest extends TestCase
138{
139    public function testMigratePreservesLocGuardThresholdsDuringMigration(): void
140    {
141        $root = sys_get_temp_dir() . '/guard-migrate-' . uniqid();
142        mkdir($root);
143        file_put_contents($root . '/loc.yaml', "scan: {roots: [lib], exclude: []}\npolicies:\n  strict:\n    limits: {file: {lines: 99}}\napply: {default: strict, rules: []}\n");
144        $config = (new \Guard\Config\Project\LegacyMigration())->migrate($root, ['version' => 1]);
145        self::assertArrayNotHasKey('scope', $config);
146        self::assertSame(['source' => ['lib'], 'exclude' => [], 'profiles' => ['strict' => ['limits' => ['file' => ['lines' => 99]]]], 'default' => 'strict', 'assignments' => []], $config['metrics']);
147    }
148
149    public function testReadPreservesLegacyMappings(): void
150    {
151        $path = tempnam(sys_get_temp_dir(), 'guard-legacy-');
152        self::assertIsString($path);
153        file_put_contents($path, 'scan: {roots: [lib]}');
154        self::assertSame(['scan' => ['roots' => ['lib']]], (new \Guard\Config\Project\LegacyMigration())->read($path));
155    }
