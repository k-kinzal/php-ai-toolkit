18/**
19 * Runs the git commands a diff site is generated from.
20 *
21 * The process launcher is injectable so the command line is verifiable
22 * without a repository, and so a failing git install produces one
23 * actionable message instead of a stack of warnings.
24 */
25final class GitCommandRunner
26{
27    /**
28     * @var Closure(string): array{status: int, output: string}
29     *
30     * @readonly
31     */
32    private Closure $launcher;
33
34    /**
35     * Creates a git runner with an injectable process launcher.
36     *
37     * @param ?Closure(string): array{status: int, output: string} $launcher
38     */
39    public function __construct(?Closure $launcher = null)
40    {
41        $this->launcher = $launcher ?? static function (string $command): array {
42            $lines = [];
43            $status = 0;
44            exec($command . ' 2>&1', $lines, $status);
45
46            return ['status' => $status, 'output' => trim(implode("\n", $lines))];
47        };
48    }
49
50    /**
51     * Runs one git command and returns its trimmed output.
52     *
53     * @param list<string> $arguments
54     *
55     * @throws DocGenException when the command fails
56     */
57    public function run(array $arguments, string $workingDirectory): string
58    {
59        $result = $this->execute($arguments, $workingDirectory);
60        if ($result['status'] !== 0) {
61            throw new DocGenException(sprintf(
62                'git %s failed in %s: %s',
63                implode(' ', $arguments),
64                $workingDirectory,
65                $result['output'] === '' ? 'no output' : $result['output'],
66            ));
67        }
68
69        return $result['output'];
70    }
71
72    /**
73     * Runs one git command and reports its status with its output.
74     *
75     * @param list<string> $arguments
76     *
77     * @return array{status: int, output: string}
78     */
79    public function execute(array $arguments, string $workingDirectory): array
80    {
81        return ($this->launcher)($this->command($arguments, $workingDirectory));
82    }
83
84    /**
85     * Builds the escaped command line of one git invocation.
86     *
87     * @param list<string> $arguments
88     */
89    public function command(array $arguments, string $workingDirectory): string
90    {
91        $parts = ['git', '-C', escapeshellarg($workingDirectory)];
92        foreach ($arguments as $argument) {
93            $parts[] = escapeshellarg($argument);
94        }
95
96        return implode(' ', $parts);
97    }
98}
