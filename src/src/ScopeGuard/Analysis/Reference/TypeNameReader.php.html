11/**
12 * Reads the class names written inside a type declaration.
13 *
14 * @visibility parent
15 */
16final class TypeNameReader
17{
18    /** @var list<string> */
19    private const RELATIVE_KEYWORDS = ['self', 'static', 'parent'];
20
21    /**
22     * Returns every class name in a type, unwrapping nullable, union, and intersection types.
23     *
24     * @return list<\PhpParser\Node\Name>
25     */
26    public function namesIn(?\PhpParser\Node $typeNode): array
27    {
28        if ($typeNode instanceof \PhpParser\Node\Name) {
29            return $this->isRelative($typeNode) ? [] : [$typeNode];
30        }
31
32        if ($typeNode instanceof \PhpParser\Node\NullableType) {
33            return $this->namesIn($typeNode->type);
34        }
35
36        if (!$typeNode instanceof \PhpParser\Node\UnionType && !$typeNode instanceof \PhpParser\Node\IntersectionType) {
37            return [];
38        }
39
40        $names = [];
41        foreach ($typeNode->types as $memberType) {
42            $names = array_merge($names, $this->namesIn($memberType));
43        }
44
45        return $names;
46    }
47
48    /**
49     * Reports whether a name is self, static, or parent.
50     *
51     * Those three can only name the class the reference is already inside or one it
52     * inherits from, and that inheritance is checked where it is declared.
53     */
54    public function isRelative(\PhpParser\Node\Name $name): bool
55    {
56        return in_array(strtolower($name->toString()), self::RELATIVE_KEYWORDS, true);
57    }
58}
