28    public function testExecuteWritesAConfigurationOnceAndRefusesToOverwrite(): void
29    {
30        $root = sys_get_temp_dir() . '/decisions-init-' . uniqid();
31        mkdir($root . '/vendor/k-kinzal/decisions-lint/rules', 0777, true);
32        file_put_contents($root . '/vendor/k-kinzal/decisions-lint/rules/cohesion.yaml', "version: 1\nrules: {}\n");
33        try {
34            $tester = new CommandTester(new InitCommand($root, $root . '/vendor/k-kinzal/decisions-lint/rules'));
35            self::assertSame(0, $tester->execute(['--provider' => 'openai', '--path' => ['app', 'lib']]));
36            self::assertStringContainsString('Wrote ' . $root . '/decisions.yaml. Export OPENAI_API_KEY', $tester->getDisplay());
37            $written = (string) file_get_contents($root . '/decisions.yaml');
38            self::assertStringContainsString("# yaml-language-server: \$schema=vendor/k-kinzal/decisions-lint/schema/decisions.schema.json\n", $written);
39            self::assertStringContainsString("provider:\n  name: openai\n  model: gpt-6-luna\n  api_key_env: OPENAI_API_KEY\n", $written);
40            self::assertStringContainsString("paths:\n  - app\n  - lib\n", $written);
41            self::assertStringContainsString("imports:\n  - vendor/k-kinzal/decisions-lint/rules/cohesion.yaml\n", $written);
42            self::assertSame(LintCommand::INVALID, $tester->execute([]));
43            self::assertStringContainsString('already exists. Edit it, or pass --config to write another file.', $tester->getDisplay());
44            self::assertSame(LintCommand::INVALID, $tester->execute(['--config' => 'other.yaml', '--provider' => 'anthropic']));
45            self::assertStringContainsString('Unsupported --provider "anthropic". Use --provider=jev, openai, ollama, localai, openai-compatible.', $tester->getDisplay());
46            self::assertSame(0, $tester->execute(['--config' => 'local.yaml', '--provider' => 'ollama']));
47            self::assertStringContainsString('Make sure a ollama server answers at http://localhost:11434, review the imported rules', $tester->getDisplay());
48            self::assertStringContainsString("provider:\n  name: ollama\n  model: llama3.1\n", (string) file_get_contents($root . '/local.yaml'));
49        } finally {
50            unlink($root . '/decisions.yaml');
51            unlink($root . '/local.yaml');
52            unlink($root . '/vendor/k-kinzal/decisions-lint/rules/cohesion.yaml');
53            rmdir($root . '/vendor/k-kinzal/decisions-lint/rules');
54            rmdir($root . '/vendor/k-kinzal/decisions-lint');
55            rmdir($root . '/vendor/k-kinzal');
56            rmdir($root . '/vendor');
57            rmdir($root);
58        }
59    }
