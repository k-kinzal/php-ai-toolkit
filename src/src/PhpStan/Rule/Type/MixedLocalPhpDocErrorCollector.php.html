40    /**
41     * Collects local errors from a statement's effective @var family.
42     *
43     * @return list<IdentifierRuleError>
44     */
45    public function errors(\PhpParser\Node\Stmt $node, string $scopeSymbol): array
46    {
47        $docComment = $node->getDocComment();
48        if ($docComment === null) {
49            return [];
50        }
51
52        $phpDoc = $this->parser->parse($docComment->getText());
53        foreach (['@phpstan-var', '@psalm-var', '@var'] as $tagName) {
54            $tags = $phpDoc->getTagsByName($tagName);
55            if ($tags === []) {
56                continue;
57            }
58
59            return $this->tagErrors($tags, $node, $scopeSymbol);
60        }
61
62        return [];
63    }
64
65    /**
66     * Collects errors from the highest-precedence tag family present.
67     *
68     * @param array<int|string, \PHPStan\PhpDocParser\Ast\PhpDoc\PhpDocTagNode> $tags
69     * @return list<IdentifierRuleError>
70     */
71    public function tagErrors(array $tags, \PhpParser\Node\Stmt $node, string $scopeSymbol): array
72    {
73        $errors = [];
74        foreach ($tags as $tag) {
75            if (!$tag->value instanceof VarTagValueNode || !$this->typeInspector->contains($tag->value->type)) {
76                continue;
77            }
78            $variableName = $tag->value->variableName;
79            $label = $variableName !== '' ? $variableName : 'expression';
80            $errors[] = $this->errorBuilder->build((string) $tag->value->type, 'local @var ' . $label, $scopeSymbol, $node->getStartLine());
81        }
82
83        return $errors;
84    }
