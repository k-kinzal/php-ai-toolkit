14/**
15 * Enforces namespace scopes declared with @visibility across analysed files.
16 *
17 * @implements Rule<CollectedDataNode>
18 */
19final class EnforceVisibilityScopeRule implements Rule
20{
21    /** @readonly */
22    private VisibilityInspector $inspector;
23
24    /** @readonly */
25    private VisibilityRuleErrorBuilder $errorBuilder;

27    /**
28     * @param list<string> $exemptNamespacePrefixes namespace subtrees allowed to cross scopes
29     */
30    public function __construct(
31        array $exemptNamespacePrefixes = ['Tests'],
32        ?VisibilityInspector $inspector = null,
33        ?VisibilityRuleErrorBuilder $errorBuilder = null,
34    ) {
35        $this->inspector = $inspector ?? new VisibilityInspector($exemptNamespacePrefixes);
36        $this->errorBuilder = $errorBuilder ?? new VisibilityRuleErrorBuilder();
37    }
38
39    /**
40     * @return class-string<CollectedDataNode>
41     */
42    public function getNodeType(): string
43    {
44        return CollectedDataNode::class;
45    }
46
47    /**
48     * @param CollectedDataNode $node
49     * @return list<IdentifierRuleError>
50     */
51    public function processNode(\PhpParser\Node $node, Scope $scope): array
52    {
53        unset($scope);
54        $index = new VisibilityDeclarationIndex();
55        foreach ($node->get(VisibilityDeclarationCollector::class) as $file => $collectedDeclarations) {
56            foreach ($collectedDeclarations as $collected) {
57                $index->add($file, $collected);
58            }
59        }
60
61        $references = [];
62        foreach ($node->get(VisibilityReferenceCollector::class) as $file => $collectedReferenceLists) {
63            foreach ($collectedReferenceLists as $collectedReferences) {
64                foreach ($collectedReferences as $reference) {
65                    $references[] = $reference + ['file' => $file];
66                }
67            }
68        }
69
70        $violations = $this->inspector->violations($index, $references);
71        usort($violations, static fn (array $left, array $right): int => [
72            $left['file'],
73            $left['line'],
74            $left['identifier'],
75            $left['symbol'],
76        ] <=> [
77            $right['file'],
78            $right['line'],
79            $right['identifier'],
80            $right['symbol'],
81        ]);
82
83        $errors = [];
84        foreach ($violations as $violation) {
85            $errors[] = $this->errorBuilder->build($violation);
86        }
87
88        return $errors;
89    }
