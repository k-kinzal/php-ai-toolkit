57    public function render(RenderKit $services, string $packageName): string
58    {
59        $pagePath = $services->url->allItemsPage($packageName);
60        $rows = $this->symbols->inPackage($services, $packageName);
61        $crumbs = [
62            ['label' => $packageName, 'path' => $services->url->packagePage($packageName)],
63            ['label' => 'All items', 'path' => null],
64        ];
65
66        return $this->chrome->page(
67            $services,
68            $pagePath,
69            'All items',
70            sprintf('Every documented item of the %s package.', $packageName),
71            $this->breadcrumb->build($services, $pagePath, $crumbs),
72            $this->sidebar->build($services, $pagePath, new SidebarScope($packageName, null, null, $this->symbolList->sections($rows))),
73            $this->content($services, $pagePath, $rows),
74        );
75    }

77    /**
