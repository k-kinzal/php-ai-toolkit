35     *
36     * @param list<PhpToken> $tokens
37     * @param list<FunctionMetric> $metrics
38     */
39    public function calculate(array $tokens, FunctionMetric $metric, array $metrics): int
40    {
41        $complexity = 1;
42        $state = new CyclomaticComplexityState();
43        for ($index = $metric->bodyStartIndex + 1; $index < $metric->bodyEndIndex; $index++) {
44            if ($this->nestedFunctionMetricRange->contains($index, $metric, $metrics)) {
45                continue;
46            }
47
48            $complexity += $this->decisionWeight->weight($tokens[$index], $state);
49            $state->advance($tokens[$index]);
50        }
51
52        return $complexity;
53    }
