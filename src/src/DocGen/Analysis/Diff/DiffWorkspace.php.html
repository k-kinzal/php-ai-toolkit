17 * Opens, analyzes, and closes the two revisions a diff site compares.

19 * A session owns its checkouts: whatever happens while it is open, the

61    /**
62     * Checks both revisions out and diffs them into one session.
63     *
64     * @param ?int $workers how many workers to analyze with, or null for the default
65     * @param ?ParseCache $cache what earlier runs already parsed, if it is kept
66     *
67     * @throws DocGenException when a revision cannot be read or analyzed
68     */
69    public function open(DocGenConfig $config, RevisionRange $range, ?int $workers = null, ?ParseCache $cache = null): DiffSession
70    {
71        $repositoryRoot = $this->repository->root($config->root);
72        $basePath = null;
73        $headPath = null;
74
75        try {
76            $basePath = $this->checkout($repositoryRoot, $range->base);
77            $baseModel = $this->analyzer->analyze($this->configFor($config, $basePath, null), $workers, $cache);
78            $headPath = $range->head !== null ? $this->checkout($repositoryRoot, $range->head) : null;
79            $headModel = $this->analyzer->analyze(
80                $headPath === null ? $config : $this->configFor($config, $headPath, $this->coverage($config)),
81                $workers,
82                $cache,
83            );
84            $index = new DiffIndex($this->label($repositoryRoot, $range->base), $this->headLabel($repositoryRoot, $range->head), $basePath);
85
86            return new DiffSession($this->differ->diff($baseModel, $headModel, $index), $index, $repositoryRoot, $basePath, $headPath);
87        } catch (DocGenException $exception) {
88            $this->removeAll($repositoryRoot, $basePath, $headPath);
89
90            throw $exception;
91        }
92    }
93
94    /**
95     * Closes one session and removes its checkouts.
96     */
97    public function close(DiffSession $session): void
98    {
99        $this->removeAll($session->repositoryRoot, $session->basePath, $session->headPath);
100    }

102    /**
103     * Checks one revision out into a linked temporary worktree.
104     *
105     * @throws DocGenException when the revision cannot be checked out
106     */
107    public function checkout(string $repositoryRoot, string $revision): string
108    {
109        $path = $this->worktree->create($repositoryRoot, $this->repository->commit($repositoryRoot, $revision));
110        $this->worktree->linkVendor($repositoryRoot, $path);
111        $canonical = realpath($path);
112
113        return $canonical === false ? $path : $canonical;
114    }
115
116    /**
117     * Removes every checkout of one session.
118     */
119    public function removeAll(string $repositoryRoot, ?string $basePath, ?string $headPath): void
120    {
121        foreach ([$basePath, $headPath] as $path) {
122            if ($path !== null) {
123                $this->worktree->remove($repositoryRoot, $path);
124            }
125        }
126    }

133     *
134     * @param string $root the checkout the revision was written to

138    {
139        return new DocGenConfig(

141            $config->packages,
142            $config->vendor,
143            $config->exclude,
144            $config->output,
145            $config->title,
146            $config->deptrac,
147            $coverage,
148            $config->vendorDev,
149            $config->cache,
150        );
151    }

153    /**
154     * Resolves the coverage report against the working tree of the project.

156    public function coverage(DocGenConfig $config): ?string
157    {
158        return $config->coverage === null ? null : $this->pathResolver->resolve($config->root, $config->coverage);

162     * Returns the display label of one revision.

169    /**

172    public function headLabel(string $repositoryRoot, ?string $revision): string
