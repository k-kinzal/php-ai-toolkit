99    /**
100     * Generates the site and optionally serves it.
101     *
102     * @param array{config: ?string, output: ?string, vendor: ?list<string>, vendorDev: ?list<string>, coverage: ?string, serve: ?string, memoryLimit: ?string, jobs: ?int, base: ?string, head: ?string, cacheDir: ?string, noCache: bool, clearCache: bool, help: bool, version: bool} $arguments
103     */
104    public function run(array $arguments): int
105    {
106        $this->memoryLimit->apply($arguments['memoryLimit']);
107
108        try {
109            $loaded = $this->loadConfig($arguments['config']);
110            $config = $this->overrides->apply($loaded, $arguments);
111            $outputRoot = $this->pathResolver->resolve($config->root, $config->output);
112            if ($arguments['clearCache']) {
113                $this->clear($config->root, $arguments['cacheDir'] ?? $loaded->cache);
114            }
115
116            $cache = $this->caches($config, $outputRoot);
117            $result = $arguments['base'] === null
118                ? $this->generate($config, $outputRoot, $arguments['jobs'], $cache)
119                : $this->generateDiff($config, $outputRoot, new RevisionRange($arguments['base'], $arguments['head']), $arguments['jobs'], $cache);
120            $this->report($result['model'], $result['pages'], $outputRoot, $cache);
121
122            if ($arguments['serve'] !== null) {
123                $this->writer->write(sprintf("Serving documentation at http://%s (Ctrl-C to stop)\n", $arguments['serve']));
124
125                return $this->previewServer->serve($outputRoot, $arguments['serve']);
126            }
127        } catch (DocGenException $exception) {
128            $this->writer->writeError(sprintf("DocGen error: %s\n", $exception->getMessage()));
129
130            return 2;
131        }
132
133        return 0;
134    }

151    /**
152     * Builds the caches of one run, or the absence of them.
153     *
154     * Both caches are read before anything is analyzed or rendered, so the
155     * worker processes of the run inherit them instead of reading the same
156     * files once per worker.
157     */
158    public function caches(DocGenConfig $config, string $outputRoot): GenerationCache
159    {
160        if ($config->cache === null) {
161            return new GenerationCache();
162        }
163
164        $directory = $this->pathResolver->resolve($config->root, $config->cache);
165        if (!$this->store->prepare($directory)) {
166            $this->writer->writeError(sprintf("Warning: nothing is cached, because the cache directory cannot be written: %s\n", $directory));
167
168            return new GenerationCache();
169        }
170
171        $cache = new GenerationCache(new ParseCache($directory), new RenderCache($directory, $outputRoot));
172        $cache->load();
173
174        return $cache;
175    }

187    /**
188     * Renders the documented project as the comparison of two revisions.
189     *
190     * The checkouts of the compared revisions are removed again whatever
191     * happens while the site is written, because they are scratch copies
192     * of the repository and nothing may outlive the run.
193     *
194     * @return array{model: ProjectModel, pages: int}
195     *
196     * @throws DocGenException when a revision cannot be read or analyzed
197     */
198    public function generateDiff(DocGenConfig $config, string $outputRoot, RevisionRange $range, ?int $workers = null, ?GenerationCache $cache = null): array
199    {
200        $cache = $cache ?? new GenerationCache();
201        $session = $this->workspace->open($config, $range, $workers, $cache->sources);
202
203        try {
204            $pages = $this->siteRenderer->render($session->model, $outputRoot, $session->diff, $workers, $cache->pages);
205            $this->writer->write(sprintf(
206                "Compared %s to %s\n",
207                $session->diff->baseLabel(),
208                $session->diff->headLabel(),
209            ));
210
211            return ['model' => $session->model, 'pages' => $pages];
212        } finally {
213            $this->workspace->close($session);
214        }
215    }

217    /**
218     * Reports what was written and what the analysis could not do.
219     */
220    public function report(ProjectModel $model, int $pages, string $outputRoot, ?GenerationCache $cache = null): void
221    {
222        $this->writer->write(sprintf(
223            "Generated %d pages for %d packages into %s\n",
224            $pages,
225            count($model->packages),
226            $outputRoot,
227        ));
228        if ($cache !== null) {
229            $this->reportCache($cache);
230        }
231
232        foreach ($model->warnings as $warning) {
233            $this->writer->writeError(sprintf("Warning: %s\n", $warning));
234        }
235    }

241    {
242        $summary = $cache->summary();
243        if ($summary !== null) {
244            $this->writer->write($summary . "\n");
245        }
246
247        $cache->save();

250    /**
251     * Loads the configuration file, or builds the zero-config defaults.
252     *
253     * @throws DocGenException when an explicitly given config file is invalid
254     */
255    public function loadConfig(?string $configOption): DocGenConfig
256    {
257        if ($configOption !== null) {
258            return $this->configLoader->load($this->configPathResolver->resolve($this->workingDirectory, $configOption));
259        }
260
261        $default = $this->configPathResolver->resolve($this->workingDirectory, 'doc.yaml');
262        if (is_file($default)) {
263            return $this->configLoader->load($default);
264        }
265
266        $root = realpath($this->workingDirectory);
267
268        return new DocGenConfig($root === false ? $this->workingDirectory : $root, ['.', 'packages/*'], [], [], 'build/docs', null, null, null);
269    }
