304    {
305        $alpha = new ClassLikeDoc('Demo\Alpha', 'Alpha', 'Demo', 'class', 'demo/app', 'src/Alpha.php', 1, 4, false, true, [], [], [], [], [], [], [], null, null, [], false);
306        $again = new ClassLikeDoc('demo\alpha', 'Alpha', 'Demo', 'class', 'demo/app', 'vendor/Alpha.php', 1, 4, false, true, [], [], [], [], [], [], [], null, null, [], false);
307        $beta = new ClassLikeDoc('Demo\Beta', 'Beta', 'Demo', 'class', 'demo/app', 'src/Beta.php', 1, 4, false, true, [], [], [], [], [], [], [], null, null, [], false);
308        $usage = new Usage('Demo\Beta', null, 'type', 'Demo\Alpha', 'run', 'src/Alpha.php', 7, false);
309
310        $merged = (new ProjectSymbolCollector())->merged([
311            [
312                ['cached' => false, 'symbols' => new FileSymbols([$alpha], []), 'usages' => [$usage]],
313                ['cached' => false, 'symbols' => 'Skipped unreadable file: src/Broken.php', 'usages' => []],
314            ],
315            [['cached' => true, 'symbols' => new FileSymbols([$again, $beta], []), 'usages' => []]],
316        ]);
317
318        self::assertSame(['Demo\Alpha', 'Demo\Beta'], [$merged['classLikes'][0]->fqcn, $merged['classLikes'][1]->fqcn]);
319        self::assertSame(['Skipped unreadable file: src/Broken.php'], $merged['warnings']);
320        self::assertSame([$usage], $merged['usages']);
321        self::assertSame([], $merged['functions']);
322    }
