15/**
16 * Locates composer.json files of installed vendor packages.
17 *
18 * Vendor directories of the project root and of every documented package are
19 * searched, so both root-level installs and per-package installs work.
20 */
21final class VendorPackageLocator
22{
23    /**
24     * Locates vendor manifest paths whose package name matches a glob.
25     *
26     * @param list<string> $searchDirectories
27     * @param list<string> $nameGlobs
28     *
29     * @return list<string>
30     */
31    public function locate(array $searchDirectories, array $nameGlobs): array
32    {
33        if ($nameGlobs === []) {
34            return [];
35        }
36
37        $paths = [];
38        foreach ($searchDirectories as $directory) {
39            $vendorDirectory = $directory . '/vendor';
40            if (!is_dir($vendorDirectory)) {
41                continue;
42            }
43
44            $manifestPaths = glob($vendorDirectory . '/*/*/composer.json');
45            foreach ($manifestPaths === false ? [] : $manifestPaths as $manifestPath) {
46                $packageDirectory = dirname($manifestPath);
47                $name = sprintf(
48                    '%s/%s',
49                    basename(dirname($packageDirectory)),
50                    basename($packageDirectory),
51                );
52                foreach ($nameGlobs as $glob) {
53                    if (fnmatch($glob, $name)) {
54                        $paths[$manifestPath] = true;
55                        break;
56                    }
57                }
58            }
59        }
60
61        $result = array_keys($paths);
62        sort($result);
63
64        return $result;
65    }
66}
