120    {
121        $guide = new MarkdownDoc('demo/pkg', 'docs/guide.md', 'docs/guide.md', 'Guide');
122        $package = new DiscoveredPackage(new ComposerManifest('/tmp/none', 'demo/pkg', 'Demo package', ['Demo\\' => ['src']], [], [], [], []), false);
123        $model = new ProjectModel('Demo Docs', '/tmp/none', [$package], new PackageGraph([]), [], [], new SymbolTable(), new HierarchyIndex(), new UsageIndex(), new TestCaseIndex(), null, [], null, [], [$guide]);

132            $html,
133        );
134        self::assertStringContainsString('<h1><span class="chip chip-kind k-document">document</span>Guide</h1>', $html);
135        self::assertStringContainsString('<h2>Guide</h2>', $html);
136    }
