296    {
297        $code = <<<'PHP'
298<?php
299
300namespace Demo;
301
302class Widget
303{
304    public function run(): int
305    {
306        return 1;
307    }
308}
309
310class Caller
311{
312    public function boot(): void
313    {
314    }
315}
316PHP;
317        $statements = (new AstParser())->parse($code, 'src/Demo/Widget.php');
318        $symbols = (new FileSymbolCollector())->collect($statements, 'demo/pkg', 'src/Demo/Widget.php', false);
319        $widget = $symbols->classLikes[0];
320        $method = $widget->methods[0];
321        $table = new SymbolTable();
322        $table->registerClassLike($symbols->classLikes[0]);
323        $table->registerClassLike($symbols->classLikes[1]);
324        $hierarchy = new HierarchyIndex();
325        $hierarchy->build($symbols->classLikes);
326        $usages = new UsageIndex();
327        $usages->build([new Usage('Demo\Widget', 'run', 'method-call', 'Demo\Caller', 'boot', 'src/Demo/Caller.php', 12, false)]);
328        $package = new DiscoveredPackage(new ComposerManifest('/tmp/none', 'demo/pkg', 'Demo package', ['Demo\\' => ['src']], [], [], [], []), false);
329        $model = new ProjectModel('Demo Docs', '/tmp/none', [$package], new PackageGraph([]), $symbols->classLikes, $symbols->functions, $table, $hierarchy, $usages, new TestCaseIndex(), null, [], null, []);
330        $services = (new SiteRenderer())->services($model);
331        $pagePath = 'demo/pkg/Demo/class.Widget.html';
332        $context = new TypeRenderContext($pagePath, 'Demo', [], [], [], $table);
333
334        $html = (new MemberHtml())->method($services, $pagePath, $widget, $method, $context);
335
336        self::assertStringContainsString('<div class="member" id="method.run">', $html);
337        self::assertStringContainsString('<pre class="member-sig"><code>', $html);
338        self::assertStringContainsString('<span class="sig-name">run</span>', $html);
339        self::assertStringContainsString('<a class="src-link" href="../../../src/src/Demo/Widget.php.html#L' . $method->startLine . '">source</a>', $html);
340        self::assertStringContainsString('<a class="anchor" href="#method.run">§</a>', $html);
341        self::assertStringContainsString('Called from <span class="count">1</span>', $html);
342        self::assertStringContainsString('>Demo\Caller::boot()</a>', $html);
343    }
