11/**
12 * Decides whether this process may fork workers.
13 *
14 * Forking is what makes parallel generation worth doing at all: a forked
15 * worker inherits the analyzed project through copy-on-write, so nothing
16 * has to be serialized to hand it the model. It is also the mechanism with
17 * the most ways to be unavailable, and every one of them has to be found
18 * before the first fork rather than after a half-written site.
19 */
20final class ForkSupport
21{
22    /**
23     * The process functions a worker pool cannot work without.
24     *
25     * @var list<string>
26     */
27    public const REQUIRED_FUNCTIONS = ['pcntl_fork', 'pcntl_waitpid', 'pcntl_wifexited', 'pcntl_wexitstatus', 'stream_socket_pair'];
28
29    /**
30     * Reports whether workers may be forked.
31     */
32    public function isAvailable(): bool
33    {
34        return $this->unavailableReason() === null;
35    }
36
37    /**
38     * Explains why workers may not be forked, or returns null when they may.
39     */
40    public function unavailableReason(): ?string
41    {
42        foreach (self::REQUIRED_FUNCTIONS as $function) {
43            if (!function_exists($function)) {
44                return 'the pcntl extension is not available';
45            }
46        }
47
48        return $this->isSharedCodeCacheEnabled() ? 'OPcache or JIT is enabled for the CLI' : null;
49    }
50
51    /**
52     * Reports whether a code cache is shared with the forked children.
53     *
54     * OPcache and the JIT keep compiled code in shared memory that is not
55     * safe to populate from several forked children at once, so a run that
56     * has either of them turned on stays sequential rather than risking a
57     * site generated from corrupted code.
58     */
59    public function isSharedCodeCacheEnabled(): bool
60    {
61        if (!function_exists('opcache_get_status')) {
62            return false;
63        }
64
65        $status = @opcache_get_status(false);
66        if ($status === false) {
67            return false;
68        }
69
70        $jit = $status['jit'] ?? null;
71
72        return ($status['opcache_enabled'] ?? false) === true || (is_array($jit) && ($jit['enabled'] ?? false) === true);
73    }
74}
