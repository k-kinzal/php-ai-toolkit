15/**
16 * Decides whether a throw inside a catch block chains the caught exception.
17 */
18final class ThrowChainEvaluator
19{
20    /** @readonly */
21    private ThrownExpression $thrown;
22
23    /**
24     * Creates the evaluator from what reads the thrown expression.
25     */
26    public function __construct(?ThrownExpression $thrown = null)
27    {
28        $this->thrown = $thrown ?? new ThrownExpression();
29    }
30
31    /**
32     * Reports whether the throw creates a new exception without referencing the caught one.
33     *
34     * Only instantiations and calls are evaluated; rethrowing a variable is
35     * never a violation because the exception object itself is preserved.
36     */
37    public function violates(Node $throw, ?string $caughtVariableName): bool
38    {
39        $expr = $this->thrown->of($throw);
40        if (!$expr instanceof CallLike) {
41            return false;
42        }
43
44        if ($caughtVariableName === null) {
45            return true;
46        }
47
48        $reference = (new NodeFinder())->findFirst(
49            $expr,
50            static fn (Node $node): bool => $node instanceof Variable
51                && is_string($node->name)
52                && $node->name === $caughtVariableName,
53        );
54
55        return $reference === null;
56    }
57}
