39    /**
40     * Writes the configuration, refusing to overwrite one.
41     *
42     * @throws LintException when the file exists, the provider is unknown, or the file cannot be written
43     */
44    #[Override]
45    protected function handle(InputInterface $input, OutputInterface $output): int
46    {
47        $path = $this->configPath($input);
48        if (file_exists($path)) {
49            throw new LintException(sprintf('%s already exists. Edit it, or pass --config to write another file.', $path));
50        }
51        $provider = $input->getOption('provider');
52        if (!is_string($provider) || !in_array($provider, ProviderSettings::NAMES, true)) {
53            throw new LintException(sprintf('Unsupported --provider "%s". Use --provider=%s.', is_string($provider) ? $provider : '', implode(', ', ProviderSettings::NAMES)));
54        }
55        $paths = [];
56        foreach (is_array($input->getOption('path')) ? $input->getOption('path') : [] as $entry) {
57            if (is_string($entry) && trim($entry) !== '') {
58                $paths[] = trim($entry);
59            }
60        }
61        $content = $this->document(ProviderSettings::defaults($provider), $paths === [] ? ['src'] : $paths, $this->imports(dirname($path)));
62        if (file_put_contents($path, $content) === false) {
63            throw new LintException(sprintf('Could not write %s. Check the directory permissions.', $path));
64        }
65        $settings = ProviderSettings::defaults($provider);
66        $key = $settings->requiresApiKey() ? sprintf('Export %s, ', $settings->apiKeyEnv) : sprintf('Make sure a %s server answers at %s, ', $provider, $settings->baseUrl);
67        $output->writeln(sprintf('Wrote %s. %sreview the imported rules with decisions-lint rules, then run decisions-lint check.', $path, $key), OutputInterface::OUTPUT_RAW);
68
69        return 0;
70    }

78    public function document(ProviderSettings $provider, array $paths, array $imports): string
79    {
80        $lines = [
81            '# yaml-language-server: $schema=' . $this->schemaPath($imports),
82            '# Nuanced rules judged by a decision model. Imported rules can be overridden below by id.',
83            'version: 1',
84            'provider:',
85            '  name: ' . $provider->name,
86            '  model: ' . $provider->model,
87            '  api_key_env: ' . $provider->apiKeyEnv,
88            'paths:',
89        ];
90        foreach ($paths as $entry) {
91            $lines[] = '  - ' . $entry;
92        }
93        $lines[] = 'exclude: []';
94        $lines[] = 'fail_on: error';
95        $lines[] = $imports === [] ? 'imports: []' : 'imports:';
96        foreach ($imports as $import) {
97            $lines[] = '  - ' . $import;
98        }
99        $lines[] = 'rules: {}';
100
101        return implode("\n", $lines) . "\n";

109    public function imports(string $directory): array
110    {
111        $rules = $this->rulesDirectory ?? dirname(__DIR__, 3) . '/rules';
112        $vendor = $directory . '/vendor/k-kinzal/decisions-lint/rules';
113        $prefix = is_dir($vendor) ? 'vendor/k-kinzal/decisions-lint/rules' : $this->relative($directory, $rules);
114        $files = glob($rules . '/*.yaml');
115        $imports = [];
116        foreach ($files === false ? [] : $files as $file) {
117            $imports[] = $prefix . '/' . basename($file);
118        }
119        sort($imports);
120
121        return $imports;
122    }

134    /**
135     * Returns a path relative to a directory, using .. where needed.
136     */
137    public function relative(string $from, string $to): string
138    {
139        $fromReal = realpath($from);
140        $toReal = realpath($to);
141        $fromParts = explode('/', trim($fromReal === false ? $from : $fromReal, '/'));
142        $toParts = explode('/', trim($toReal === false ? $to : $toReal, '/'));
143        while ($fromParts !== [] && $toParts !== [] && $fromParts[0] === $toParts[0]) {
144            array_shift($fromParts);
145            array_shift($toParts);
146        }
147
148        return implode('/', array_merge(array_fill(0, count($fromParts), '..'), $toParts));
